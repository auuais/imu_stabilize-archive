// ======= Simple profiler =======
using clock_hr = std::chrono::high_resolution_clock;
using ms_f = std::chrono::duration<double, std::milli>;

struct StageStat {
    const char* name;
    int   calls = 0;
    double ms_total = 0.0;
    double bytes_r = 0.0;
    double bytes_w = 0.0;
};
enum StageId {
    STG_TOTAL = 0,
    STG_IO,
    STG_MAPS,
    STG_UNDMAP,
    STG_LUT,
    STG_RENDER,
    STG_BLEND,
    STG__COUNT
};
static StageStat G_STATS[STG__COUNT] = {
    {"Total"}, {"I/O"}, {"Maps"}, {"UndistMap"}, {"LUT"}, {"Render"}, {"Blend"}
};

struct StageTimer {
    StageId id;
    clock_hr::time_point t0;
    double br = 0, bw = 0;
    StageTimer(StageId i, double bytes_r = 0, double bytes_w = 0) : id(i), t0(clock_hr::now()), br(bytes_r), bw(bytes_w) {}
    ~StageTimer() {
        auto t1 = clock_hr::now();
        double ms = ms_f(t1 - t0).count();
        G_STATS[id].calls++;
        G_STATS[id].ms_total += ms;
        G_STATS[id].bytes_r += br;
        G_STATS[id].bytes_w += bw;
    }
};
static void print_profile(const char* title) {
    printf("\n===== %s =====\n", title);
    for (int i = 0; i < STG__COUNT; ++i) {
        if (G_STATS[i].calls == 0) continue;
        double rps = (G_STATS[i].ms_total > 0) ? (G_STATS[i].bytes_r / 1e6) / (G_STATS[i].ms_total / 1000.0) : 0;
        double wps = (G_STATS[i].ms_total > 0) ? (G_STATS[i].bytes_w / 1e6) / (G_STATS[i].ms_total / 1000.0) : 0;
        printf("Stage %-10s | calls:%4d | time:%10.3f ms | R:%8.2f MB | W:%8.2f MB | R:%8.2f MB/s | W:%8.2f MB/s\n",
            G_STATS[i].name, G_STATS[i].calls, G_STATS[i].ms_total,
            G_STATS[i].bytes_r / 1e6, G_STATS[i].bytes_w / 1e6, rps, wps);
    }
    // FPS from TOTAL if present
    if (G_STATS[STG_TOTAL].ms_total > 0) {
        double ms = G_STATS[STG_TOTAL].ms_total;
        printf("\nTOTAL wall time: %.3f ms  (%.2f FPS)\n", ms, 1000.0 / ms);
    }
}
static void reset_profile() { for (int i = 0; i < STG__COUNT; ++i) { G_STATS[i].calls = 0; G_STATS[i].ms_total = 0; G_STATS[i].bytes_r = 0; G_STATS[i].bytes_w = 0; } }

#define PIXEL(img, y, x, c) ((img)->data[((y) * (img)->width + (x)) * (img)->channels + (c)])

static unsigned char saturate_cast(float val) {
    if (val < 0) return 0;
    if (val > 255) return 255;
    return (unsigned char)(val + 0.5f);
}

// ===== BMP IO =====
Image* readBMP(const char* filename) {
    FILE* fp = fopen(filename, "rb");
    if (!fp) return nullptr;
    CustomBitmapFileHeader fileHeader;
    CustomBitmapInfoHeader infoHeader;
    fread(&fileHeader, sizeof(BITMAPFILEHEADER), 1, fp);
    fread(&infoHeader, sizeof(BITMAPINFOHEADER), 1, fp);
    Image* img = (Image*)malloc(sizeof(Image));
    img->width = infoHeader.biWidth;
    img->height = abs(infoHeader.biHeight);
    img->channels = 3;
    int row_padded = ((img->width * 3 + 3) / 4) * 4;
    img->data = (unsigned char*)malloc(img->width * img->height * 3);
    fseek(fp, fileHeader.bfOffBits, SEEK_SET);
    for (int i = 0; i < img->height; i++) {
        fread(img->data + i * img->width * 3, 1, img->width * 3, fp);
        fseek(fp, row_padded - img->width * 3, SEEK_CUR);
    }
    fclose(fp);
    return img;
}
int writeBMP(const char* filename, const Image* img) {
    FILE* fp = fopen(filename, "wb");
    if (!fp) return 0;
    int row_padded = ((img->width * 3 + 3) / 4) * 4;
    int fileSize = 54 + row_padded * img->height;
    unsigned char bmpfileheader[14] = { 'B','M', 0,0,0,0, 0,0, 0,0, 54,0,0,0 };
    unsigned char bmpinfoheader[40] = { 0 };
    unsigned char bmppad[3] = { 0,0,0 };
    bmpfileheader[2] = (unsigned char)(fileSize);
    bmpfileheader[3] = (unsigned char)(fileSize >> 8);
    bmpfileheader[4] = (unsigned char)(fileSize >> 16);
    bmpfileheader[5] = (unsigned char)(fileSize >> 24);
    bmpinfoheader[0] = 40;
    bmpinfoheader[4] = (unsigned char)(img->width);
    bmpinfoheader[5] = (unsigned char)(img->width >> 8);
    bmpinfoheader[6] = (unsigned char)(img->width >> 16);
    bmpinfoheader[7] = (unsigned char)(img->width >> 24);
    bmpinfoheader[8] = (unsigned char)(img->height);
    bmpinfoheader[9] = (unsigned char)(img->height >> 8);
    bmpinfoheader[10] = (unsigned char)(img->height >> 16);
    bmpinfoheader[11] = (unsigned char)(img->height >> 24);
    bmpinfoheader[12] = 1;
    bmpinfoheader[14] = 24;
    fwrite(bmpfileheader, 1, 14, fp);
    fwrite(bmpinfoheader, 1, 40, fp);
    for (int i = 0; i < img->height; ++i) {
        fwrite(img->data + i * img->width * 3, 1, img->width * 3, fp);
        fwrite(bmppad, 1, row_padded - img->width * 3, fp);
    }
    fclose(fp);
    return 1;
}
void free_image(Image* img) {
    if (!img) return;
    free(img->data);
    free(img);
}

// ===== Simple bilinear (used only for map-of-map float sampling) =====
void interpolation(const Image* img, float x, float y, unsigned char* result) {
    int x0 = (int)floorf(x);
    int y0 = (int)floorf(y);
    float dx = x - x0;
    float dy = y - y0;
    for (int c = 0; c < 3; ++c) result[c] = 0;
    if (x0 < 0 || x0 >= img->width - 1 || y0 < 0 || y0 >= img->height - 1) return;
    for (int c = 0; c < 3; ++c) {
        float p00 = PIXEL(img, y0, x0, c);
        float p10 = PIXEL(img, y0, x0 + 1, c);
        float p01 = PIXEL(img, y0 + 1, x0, c);
        float p11 = PIXEL(img, y0 + 1, x0 + 1, c);
        float val = (1 - dx) * (1 - dy) * p00 + dx * (1 - dy) * p10 + (1 - dx) * dy * p01 + dx * dy * p11;
        result[c] = (unsigned char)(val > 255 ? 255 : (val < 0 ? 0 : val));
    }
}

    *map_x = (Image*)malloc(sizeof(Image));
    *map_y = (Image*)malloc(sizeof(Image));

    (*map_y)->width = width; (*map_y)->height = height; (*map_y)->channels = 1;
    (*map_x)->data = (unsigned char*)malloc(width * height * sizeof(float));
    (*map_y)->data = (unsigned char*)malloc(width * height * sizeof(float));
    std::filesystem::path mapx_path = exePath.parent_path() / "map_x.bin";
    std::filesystem::path mapy_path = exePath.parent_path() / "map_y.bin";
    FILE* fx = fopen(mapx_path.string().c_str(), "rb");
    fread((*map_x)->data, sizeof(float), width * height, fx); fclose(fx);
    FILE* fy = fopen(mapy_path.string().c_str(), "rb");
    fread((*map_y)->data, sizeof(float), width * height, fy); fclose(fy);
    printf("[INFO] Cylindrical maps loaded from disk.\n");
}
void generateCylindricalRemap(Image** map_x, Image** map_y, int width, int height, float fov_h_deg, float fov_v_deg, float x_shift, const std::filesystem::path& exePath) {
    float fov_h_rad = fov_h_deg * (M_PI / 180.0f);
    float fov_v_rad = fov_v_deg * (M_PI / 180.0f);
    float fx_cyl = width / (2.0f * tanf(fov_h_rad / 2.0f));
    float fy_cyl = height / (2.0f * tanf(fov_v_rad / 2.0f));
    float cx_cyl = width / 2.0f;
    float cy_cyl = height / 2.0f;
    *map_x = (Image*)malloc(sizeof(Image));
    *map_y = (Image*)malloc(sizeof(Image));
    (*map_x)->width = width;  (*map_x)->height = height;  (*map_x)->channels = 1;
    (*map_y)->width = width;  (*map_y)->height = height;  (*map_y)->channels = 1;
    (*map_x)->data = (unsigned char*)malloc(width * height * sizeof(float));
    (*map_y)->data = (unsigned char*)malloc(width * height * sizeof(float));
    for (int y = 0; y < height; ++y) {
        for (int x = 0; x < width; ++x) {
            float theta = (x - cx_cyl + x_shift) / fx_cyl;
            float h = (y - cy_cyl) / fy_cyl;
            float x_source = sinf(theta);
            float y_source = h;
            float z_source = cosf(theta);
            float src_x = fx * resize_factor * (x_source / z_source) + cx * resize_factor;
            float src_y = fy * resize_factor * (y_source / z_source) + cy * resize_factor;
            int index = y * width + x;
            ((float*)(*map_x)->data)[index] = src_x;
            ((float*)(*map_y)->data)[index] = src_y;
        }
    }
    std::filesystem::path mapx_path = exePath.parent_path() / "map_x.bin";
    std::filesystem::path mapy_path = exePath.parent_path() / "map_y.bin";
    FILE* fxp = fopen(mapx_path.string().c_str(), "wb");
    fwrite((*map_x)->data, sizeof(float), width * height, fxp); fclose(fxp);
    FILE* fyp = fopen(mapy_path.string().c_str(), "wb");
    fwrite((*map_y)->data, sizeof(float), width * height, fyp); fclose(fyp);
    printf("[INFO] Cylindrical maps saved to %s and %s\n", mapx_path.string().c_str(), mapy_path.string().c_str());
}

// ===== Undistort map (maps undistorted pixel -> distorted sensor coords) =====
void generateUndistortMap(
    int width, int height,
    float fx_, float fy_, float cx_, float cy_,
    float fxn, float fyn, float cxn, float cyn,
    float k1, float k2, float k3, float p1, float p2,
    Image** map_x, Image** map_y)
{
    *map_x = (Image*)malloc(sizeof(Image));
    *map_y = (Image*)malloc(sizeof(Image));
    (*map_x)->width = width;  (*map_x)->height = height;  (*map_x)->channels = 1;
    (*map_y)->width = width;  (*map_y)->height = height;  (*map_y)->channels = 1;
    (*map_x)->data = (unsigned char*)malloc(width * height * sizeof(float));
    (*map_y)->data = (unsigned char*)malloc(width * height * sizeof(float));
    for (int v = 0; v < height; ++v) {
        for (int u = 0; u < width; ++u) {
            float x = (u - cxn) / fxn;
            float y = (v - cyn) / fyn;
            float r2 = x * x + y * y;
            float radial = 1 + k1 * r2 + k2 * r2 * r2 + k3 * r2 * r2 * r2;
            float dx = 2 * p1 * x * y + p2 * (r2 + 2 * x * x);
            float dy = p1 * (r2 + 2 * y * y) + 2 * p2 * x * y;
            float x_dist = x * radial + dx;
            float y_dist = y * radial + dy;
            float ud = x_dist * fx_ + cx_;
            float vd = y_dist * fy_ + cy_;
            int idx = v * width + u;
            ((float*)(*map_x)->data)[idx] = ud;
            ((float*)(*map_y)->data)[idx] = vd;
        }
    }
    printf("[INFO] Undistortion maps created. (%dx%d)\n", width, height);
}

// ===== Fixed-point bilinear line-buffer renderer =====
#define WFIX_SHIFT 16
#define WFIX_ONE   (1u << WFIX_SHIFT)

struct MapEntry {
    int32_t x0;
    int32_t y0;
    uint16_t wx; // frac(x) in Q0.16
    uint16_t wy; // frac(y) in Q0.16
};
struct CompositeMap {
    int out_w, out_h;
    std::vector<MapEntry> lut; // out_w*out_h
};

static inline void inverse_affine_point(
    float x, float y,
    float cx, float cy,
    float angle_rad, float y_shift,
    float& x_out, float& y_out)
{
    // forward had: rotate + (-y_shift); here we invert that
    float dy = y - cy;
    dy += y_shift; // inverse of -y_shift
    float dx = x - cx;
    float cos_a = cosf(-angle_rad);
    float sin_a = sinf(-angle_rad);
    float src_x = cos_a * dx + sin_a * dy + cx;
    float src_y = -sin_a * dx + cos_a * dy + cy;
    x_out = src_x; y_out = src_y;
}

// Build composite LUT for one camera / one frame
static void buildCompositeMapForCam(
    const Image* map_cyl_x, const Image* map_cyl_y, // resized space maps
    int width_r, int height_r,
    int out_w, int out_h,
    int crop_x0, int crop_y0,
    float roll_deg, float y_shift,
    float resize_factor_local,
#if USE_UNDISTORT_IN_LUT
    const Image* und_map_x, const Image* und_map_y,
#endif
    CompositeMap& out_map)
{
    out_map.out_w = out_w;
    out_map.out_h = out_h;
    out_map.lut.resize((size_t)out_w * out_h);

    const float cx_r = width_r * 0.5f;
    const float cy_r = height_r * 0.5f;
    const float angle = roll_deg * (float)M_PI / 180.0f;

    for (int y = 0; y < out_h; ++y) {
        for (int x = 0; x < out_w; ++x) {
            // 1) back to resized space (undo crop)
            float xr = (float)(x + crop_x0);
            float yr = (float)(y + crop_y0);

            // 2) inverse affine (output -> pre-affine resized cyl space)
            float xa, ya;
            inverse_affine_point(xr, yr, cx_r, cy_r, angle, y_shift, xa, ya);

            // 3) map-of-map sample cylindrical maps at (xa,ya)
            int x0m = (int)floorf(xa), y0m = (int)floorf(ya);
            float fxm = xa - x0m, fym = ya - y0m;
            if (x0m < 0) { x0m = 0; fxm = 0; }
            if (y0m < 0) { y0m = 0; fym = 0; }
            if (x0m >= width_r - 1) { x0m = width_r - 2; fxm = 1; }
            if (y0m >= height_r - 1) { y0m = height_r - 2; fym = 1; }
            int idx00 = y0m * width_r + x0m;
            int idx10 = idx00 + 1;
            int idx01 = idx00 + width_r;
            int idx11 = idx01 + 1;

            float mx00 = ((float*)map_cyl_x->data)[idx00];
            float mx10 = ((float*)map_cyl_x->data)[idx10];
            float mx01 = ((float*)map_cyl_x->data)[idx01];
            float mx11 = ((float*)map_cyl_x->data)[idx11];

            float my00 = ((float*)map_cyl_y->data)[idx00];
            float my10 = ((float*)map_cyl_y->data)[idx10];
            float my01 = ((float*)map_cyl_y->data)[idx01];
            float my11 = ((float*)map_cyl_y->data)[idx11];

            float src_x_resized =
                (1 - fxm) * (1 - fym) * mx00 + fxm * (1 - fym) * mx10 +
                (1 - fxm) * fym * mx01 + fxm * fym * mx11;
            float src_y_resized =
                (1 - fxm) * (1 - fym) * my00 + fxm * (1 - fym) * my10 +
                (1 - fxm) * fym * my01 + fxm * fym * my11;

            // 4) back to ORIGINAL undistorted space
            float su = src_x_resized / resize_factor_local;
            float sv = src_y_resized / resize_factor_local;

#if USE_UNDISTORT_IN_LUT
            // 5) fold lens distortion: map (su,sv) in undistorted space to distorted sensor coords via undistort map-of-map
            int ux0 = (int)floorf(su), uy0 = (int)floorf(sv);
            float ufx = su - ux0, ufy = sv - uy0;
            if (ux0 < 0) { ux0 = 0; ufx = 0; }
            if (uy0 < 0) { uy0 = 0; ufy = 0; }
            if (ux0 >= WIDTH - 1) { ux0 = WIDTH - 2; ufx = 1; }
            if (uy0 >= HEIGHT - 1) { uy0 = HEIGHT - 2; ufy = 1; }
            int uidx00 = uy0 * WIDTH + ux0;
            int uidx10 = uidx00 + 1;
            int uidx01 = uidx00 + WIDTH;
            int uidx11 = uidx01 + 1;
            float dx00 = ((float*)und_map_x->data)[uidx00];
            float dx10 = ((float*)und_map_x->data)[uidx10];
            float dx01 = ((float*)und_map_x->data)[uidx01];
            float dx11 = ((float*)und_map_x->data)[uidx11];
            float dy00 = ((float*)und_map_y->data)[uidx00];
            float dy10 = ((float*)und_map_y->data)[uidx10];
            float dy01 = ((float*)und_map_y->data)[uidx01];
            float dy11 = ((float*)und_map_y->data)[uidx11];
            float src_x = (1 - ufx) * (1 - ufy) * dx00 + ufx * (1 - ufy) * dx10 +
                (1 - ufx) * ufy * dx01 + ufx * ufy * dx11;
            float src_y = (1 - ufx) * (1 - ufy) * dy00 + ufx * (1 - ufy) * dy10 +
                (1 - ufx) * ufy * dy01 + ufx * ufy * dy11;
#else
            float src_x = su, src_y = sv; // sample directly from undistorted image buffer (if you had it)
#endif

            // 6) store (x0,y0) and fixed-point weights for bilinear
            int32_t sx0 = (int32_t)floorf(src_x);
            int32_t sy0 = (int32_t)floorf(src_y);
            float wxf = src_x - (float)sx0;
            float wyf = src_y - (float)sy0;

            if (sx0 < 0) { sx0 = 0; wxf = 0; }
            if (sy0 < 0) { sy0 = 0; wyf = 0; }
            if (sx0 >= WIDTH - 1) { sx0 = WIDTH - 2; wxf = 1; }
            if (sy0 >= HEIGHT - 1) { sy0 = HEIGHT - 2; wyf = 1; }

            MapEntry me;
            me.x0 = sx0;
            me.y0 = sy0;
            uint32_t wxi = (uint32_t)lroundf(wxf * (float)WFIX_ONE);
            uint32_t wyi = (uint32_t)lroundf(wyf * (float)WFIX_ONE);
            if (wxi > 0xFFFF) wxi = 0xFFFF;
            if (wyi > 0xFFFF) wyi = 0xFFFF;
            me.wx = (uint16_t)wxi;
            me.wy = (uint16_t)wyi;
            out_map.lut[(size_t)y * out_w + x] = me;
        }
    }
}

// Two-line-buffer renderer using the composite LUT
static Image* renderWithLineBuffer(const Image* src, const CompositeMap& cmap) {
    const int out_w = cmap.out_w, out_h = cmap.out_h;
    Image* out = (Image*)malloc(sizeof(Image));
    out->width = out_w; out->height = out_h; out->channels = 3;
    out->data = (unsigned char*)calloc(out_w * out_h * 3, 1);

    for (int y = 0; y < out_h; ++y) {
        const MapEntry* row = &cmap.lut[(size_t)y * out_w];
        for (int x = 0; x < out_w; ++x) {
            const MapEntry& me = row[x];
            int x0 = me.x0, y0 = me.y0;
            if (x0 < 0) x0 = 0;
            if (y0 < 0) y0 = 0;
            if (x0 >= src->width - 1)  x0 = src->width - 2;
            if (y0 >= src->height - 1) y0 = src->height - 2;

            const uint32_t ix = me.wx, iy = me.wy;
            const uint32_t ixn = WFIX_ONE - ix, iyn = WFIX_ONE - iy;

            const unsigned char* row0 = &src->data[(y0 * src->width + x0) * 3];
            const unsigned char* row1 = &src->data[((y0 + 1) * src->width + x0) * 3];

            unsigned char* dstp = &out->data[(y * out_w + x) * 3];
            for (int c = 0; c < 3; ++c) {
                uint32_t p00 = row0[c];
                uint32_t p10 = row0[c + 3];
                uint32_t p01 = row1[c];
                uint32_t p11 = row1[c + 3];
                uint64_t w00 = (uint64_t)ixn * (uint64_t)iyn;
                uint64_t w10 = (uint64_t)ix * (uint64_t)iyn;
                uint64_t w01 = (uint64_t)ixn * (uint64_t)iy;
                uint64_t w11 = (uint64_t)ix * (uint64_t)iy;
                uint64_t acc = w00 * p00 + w10 * p10 + w01 * p01 + w11 * p11; // Q32
                acc >>= 32;
                if (acc > 255) acc = 255;
                dstp[c] = (unsigned char)acc;
            }
        }
    }
    return out;
}

// ===== Feather blend & resize-to-fit (unchanged) =====
Image* ResizeImage(const Image* input, float scale) {
    int new_width = (int)(input->width * scale);
    int new_height = (int)(input->height * scale);
    Image* output = (Image*)malloc(sizeof(Image));
    output->width = new_width; output->height = new_height; output->channels = input->channels;
    output->data = (unsigned char*)calloc(new_width * new_height * input->channels, 1);
    for (int y = 0; y < new_height; ++y) {
        for (int x = 0; x < new_width; ++x) {
            float src_x = (float)x / scale;
            float src_y = (float)y / scale;
            unsigned char pixel[3];
            interpolation(input, src_x, src_y, pixel);
            for (int c = 0; c < input->channels; ++c)
                output->data[(y * new_width + x) * input->channels + c] = pixel[c];
        }
    }
    return output;
}
Image* featherBlendFit(const std::vector<Image*>& frames, int overlap, int target_width, int target_height) {
    if (frames.empty()) return NULL;
    int height = frames[0]->height, channels = frames[0]->channels;
    int total_width = frames[0]->width;
    for (int i = 1; i < (int)frames.size(); ++i) total_width += (frames[i]->width - overlap);
    Image* blended = (Image*)malloc(sizeof(Image));
    blended->width = total_width; blended->height = height; blended->channels = channels;
    blended->data = (unsigned char*)calloc(total_width * height * channels, 1);
    int current_x = 0;
    for (int i = 0; i < (int)frames.size(); ++i) {
        Image* img = frames[i];
        for (int y = 0; y < height; ++y) {
            for (int x = 0; x < img->width; ++x) {
                int out_x = current_x + x;
                if (i > 0 && x < overlap) {
                    float alpha = (float)x / (float)overlap;
                    for (int c = 0; c < channels; ++c) {
                        unsigned char prev = PIXEL(blended, y, out_x, c);
                        unsigned char curr = PIXEL(img, y, x, c);
                        PIXEL(blended, y, out_x, c) = (unsigned char)((1 - alpha) * prev + alpha * curr);
                    }
                }
                else {
                    for (int c = 0; c < channels; ++c)
                        PIXEL(blended, y, out_x, c) = PIXEL(img, y, x, c);
                }
            }
        }
        current_x += (img->width - overlap);
    }
    float scale = (float)target_width / blended->width;
    Image* resized = ResizeImage(blended, scale);
    free(blended->data); free(blended);
    Image* final_output = (Image*)malloc(sizeof(Image));
    int new_height = (int)(height * scale);
    final_output->width = target_width; final_output->height = target_height; final_output->channels = channels;
    final_output->data = (unsigned char*)calloc(target_width * target_height * channels, 1);
    int y_offset = (target_height - resized->height) / 2;
    for (int y = 0; y < resized->height; ++y)
        for (int x = 0; x < target_width; ++x)
            for (int c = 0; c < channels; ++c)
                PIXEL(final_output, y + y_offset, x, c) = PIXEL(resized, y, x, c);
    free(resized->data); free(resized);
    return final_output;
}

// ===== Utilities =====
std::wstring StringToWString(const std::string& str) {
    if (str.empty()) return {};
    int size_needed = MultiByteToWideChar(CP_UTF8, 0, str.data(), (int)str.size(), NULL, 0);
    std::wstring result(size_needed, 0);
    MultiByteToWideChar(CP_UTF8, 0, str.data(), (int)str.size(), &result[0], size_needed);
    return result;
}
std::string WStringToUtf8(const std::wstring& wstr) {
    if (wstr.empty()) return {};
    int size = WideCharToMultiByte(CP_UTF8, 0, wstr.data(), (int)wstr.length(), nullptr, 0, nullptr, nullptr);
    std::string result(size, 0);
    WideCharToMultiByte(CP_UTF8, 0, wstr.data(), (int)wstr.length(), &result[0], size, nullptr, nullptr);
    return result;
}
std::string SaveImageFilePath(const std::wstring& captureDir, const std::wstring& prefix) {
    SYSTEMTIME st; GetLocalTime(&st);
    wchar_t buffer[512];
    swprintf_s(buffer, 512, L"%s\\%04d%02d%02d_%02d%02d%02d_%s.bmp",
        captureDir.c_str(), st.wYear, st.wMonth, st.wDay, st.wHour, st.wMinute, st.wSecond, prefix.c_str());
    std::wstring widePath(buffer);
    return WStringToUtf8(widePath);
}
float clamp5(float v) { return (v > 5.0f) ? 5.0f : ((v < -5.0f) ? -5.0f : v); }

void convert_rpy_sensor_to_cam(
    double roll_sys1, double pitch_sys1, double yaw_offset,
    double pitch_offset, double roll_offset,
    float* roll_deg_out, float* pitch_deg_out, float* yaw_deg_out)
{
    double roll_rad = roll_sys1 * M_PI / 180.0;
    double pitch_rad = pitch_sys1 * M_PI / 180.0;
    double roll_offset_rad = roll_offset * M_PI / 180.0;
    double pitch_offset_rad = pitch_offset * M_PI / 180.0;
    double yaw_offset_rad = yaw_offset * M_PI / 180.0;
    double Rx[3][3] = { {1,0,0},{0,cos(roll_rad),-sin(roll_rad)},{0,sin(roll_rad),cos(roll_rad)} };
    double Ry[3][3] = { {cos(pitch_rad),0,sin(pitch_rad)},{0,1,0},{-sin(pitch_rad),0,cos(pitch_rad)} };
    double R_sys1[3][3];
    for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j)
        R_sys1[i][j] = Rx[i][0] * Ry[0][j] + Rx[i][1] * Ry[1][j] + Rx[i][2] * Ry[2][j];
    double R_roll_offset[3][3] = { {1,0,0},{0,cos(roll_offset_rad),-sin(roll_offset_rad)},{0,sin(roll_offset_rad),cos(roll_offset_rad)} };
    double R_pitch_offset[3][3] = { {cos(pitch_offset_rad),0,sin(pitch_offset_rad)},{0,1,0},{-sin(pitch_offset_rad),0,cos(pitch_offset_rad)} };
    double R_yaw_offset[3][3] = { {cos(yaw_offset_rad),-sin(yaw_offset_rad),0},{sin(yaw_offset_rad),cos(yaw_offset_rad),0},{0,0,1} };
    double R_temp[3][3], R_offset[3][3];
    for (int i = 0; i < 3; ++i)for (int j = 0; j < 3; ++j)
        R_temp[i][j] = R_yaw_offset[i][0] * R_pitch_offset[0][j] + R_yaw_offset[i][1] * R_pitch_offset[1][j] + R_yaw_offset[i][2] * R_pitch_offset[2][j];
    for (int i = 0; i < 3; ++i)for (int j = 0; j < 3; ++j)
        R_offset[i][j] = R_temp[i][0] * R_roll_offset[0][j] + R_temp[i][1] * R_roll_offset[1][j] + R_temp[i][2] * R_roll_offset[2][j];
    double R_sysX[3][3];
    for (int i = 0; i < 3; ++i)for (int j = 0; j < 3; ++j)
        R_sysX[i][j] = R_sys1[i][0] * R_offset[0][j] + R_sys1[i][1] * R_offset[1][j] + R_sys1[i][2] * R_offset[2][j];
    double pitchX = asin(-R_sysX[2][0]);
    double cy = cos(pitchX);
    double rollX, yawX;
    if (fabs(cy) > 1e-6) {
        rollX = atan2(R_sysX[2][1], R_sysX[2][2]);
        yawX = atan2(R_sysX[1][0], R_sysX[0][0]);
    }
    else {
        rollX = 0;
        yawX = atan2(-R_sysX[0][1], R_sysX[1][1]);
    }
    *roll_deg_out = (float)(rollX * 180.0 / M_PI);
    *pitch_deg_out = (float)(pitchX * 180.0 / M_PI);
    *yaw_deg_out = (float)(yawX * 180.0 / M_PI);
}

// =============================== MAIN ===============================
int main() {
    StageTimer T_total(STG_TOTAL); // whole app wall time

    // Resolve paths
    wchar_t szPath[MAX_PATH] = L"";
    GetModuleFileNameW(NULL, szPath, MAX_PATH);
    std::filesystem::path exePath(szPath);
    std::filesystem::path imuIniPath = exePath; imuIniPath.replace_extension(L".ini");
    std::filesystem::path paramsIniPath = exePath; paramsIniPath.replace_filename(L"parameters.ini");

    INIManager imuIni(imuIniPath);
    INIManager paramIni(paramsIniPath);

    // Read parameters
    WIDTH = paramIni.ReadInteger(L"image", L"width", 1920);
    HEIGHT = paramIni.ReadInteger(L"image", L"height", 1080);
    NUM_CAMS = paramIni.ReadInteger(L"image", L"num_cams", 6);
    BUFFER_LINES = paramIni.ReadInteger(L"image", L"buffer_lines", 15);

    fx = paramIni.ReadFloat(L"calibration", L"fx", 1204.505857f);
    fy = paramIni.ReadFloat(L"calibration", L"fy", 1204.505857f);
    float cx_ratio = paramIni.ReadFloat(L"calibration", L"cx_ratio", 0.5f);
    float cy_ratio = paramIni.ReadFloat(L"calibration", L"cy_ratio", 0.5f);
    cx = WIDTH * cx_ratio; cy = HEIGHT * cy_ratio;

    fx_new = paramIni.ReadFloat(L"calibration", L"fx_dist", 1204.505857f);
    fy_new = paramIni.ReadFloat(L"calibration", L"fy_dist", 1204.505857f);
    cx_new = paramIni.ReadFloat(L"calibration", L"cx_dist", 947.559f);
    cy_new = paramIni.ReadFloat(L"calibration", L"cy_dist", 534.965f);

    K1 = paramIni.ReadFloat(L"calibration", L"k1", -0.343698f);
    K2 = paramIni.ReadFloat(L"calibration", L"k2", 0.105503f);
    K3 = paramIni.ReadFloat(L"calibration", L"k3", 0.0f);
    P1 = paramIni.ReadFloat(L"calibration", L"p1", -0.001558f);
    P2 = paramIni.ReadFloat(L"calibration", L"p2", -0.002416f);

    cylindrical_H_fov_deg = paramIni.ReadFloat(L"projection", L"cylindrical_H_fov_deg", 60.0f);
    cylindrical_V_fov_deg = paramIni.ReadFloat(L"projection", L"cylindrical_V_fov_deg", 51.3f);

    panorama_crop_height_scale = paramIni.ReadFloat(L"cropping", L"panorama_crop_height_scale", 0.80f);
    panorama_crop_width_scale = paramIni.ReadFloat(L"cropping", L"panorama_crop_width_scale", 0.80f);
    original_crop_height_scale = paramIni.ReadFloat(L"cropping", L"original_crop_height_scale", 0.70f);
    original_crop_width_scale = paramIni.ReadFloat(L"cropping", L"original_crop_width_scale", 0.54f);

    resize_factor = paramIni.ReadFloat(L"resize", L"resize_factor ", 0.5f);

    PITCH_TO_TRANSLATION_FACTOR = paramIni.ReadFloat(L"transform", L"pitch_to_translation_factor", -17.44153917818162f);
    PITCH_TO_TRANSLATION_FACTOR *= resize_factor;
    YAW_TO_TRANSLATION_FACTOR = paramIni.ReadFloat(L"transform", L"yaw_to_translation_factor", 17.84f);

    overlap_px = paramIni.ReadInteger(L"blend", L"overlap_px ", 100);

    // IMU ini and capture dir
    PathRemoveFileSpecW(szPath);
    std::wstring dirCapture = std::wstring(szPath) + L"\\Capture";
    CreateDirectoryW(dirCapture.c_str(), NULL);

    std::vector<std::string> filenames;
    for (int i = 0; i < NUM_CAMS; ++i) {
        std::wstring key = L"file_" + std::to_wstring(i);
        std::wstring val = imuIni.ReadString(L"images", key, L"");
        filenames.push_back(WStringToUtf8(val));
    }
    std::vector<float> yaw_offset_cam = imuIni.ReadFloatArray(L"offset", L"yaw");
    std::vector<float> pitch_offset_cam = imuIni.ReadFloatArray(L"offset", L"pitch");
    std::vector<float> roll_offset_cam = imuIni.ReadFloatArray(L"offset", L"roll");

    Stabilization = imuIni.ReadBoolean(L"Sensor_Orientation", L"stabilization", true);
    if (Stabilization) {
        roll_Sensor = clamp5(imuIni.ReadFloat(L"Sensor_Orientation", L"roll", 0.0f));
        pitch_Sensor = clamp5(imuIni.ReadFloat(L"Sensor_Orientation", L"pitch", 0.0f));
        yaw_Sensor = clamp5(imuIni.ReadFloat(L"Sensor_Orientation", L"yaw", 0.0f));
    }
    else {
        roll_Sensor = pitch_Sensor = yaw_Sensor = 0.0f;
    }

    printf("\n........Processing........\n");

    // Load/generate maps (profiled)
    Image* map_x = nullptr, * map_y = nullptr;
    std::filesystem::path mapx_path = exePath.parent_path() / "map_x.bin";
    std::filesystem::path mapy_path = exePath.parent_path() / "map_y.bin";
    int width_r = (int)(WIDTH * resize_factor);
    int height_r = (int)(HEIGHT * resize_factor);

    {
        // maps are ~ 2 * width_r * height_r * sizeof(float) reads
        double br = (double)width_r * (double)height_r * sizeof(float) * 2.0;
        StageTimer T(STG_MAPS, br, 0);
        if (std::filesystem::exists(mapx_path) && std::filesystem::exists(mapy_path)) {
            loadCylindricalMaps(&map_x, &map_y, width_r, height_r, exePath);
        }
        else {
            // writes same size back to disk once
            double bw = br;
            // (nesting a second timer for generation's compute+write)
            StageTimer T2(STG_MAPS, 0, bw);
            generateCylindricalRemap(&map_x, &map_y, width_r, height_r, cylindrical_H_fov_deg, cylindrical_V_fov_deg, 0.0f, exePath);
        }
    }

#if USE_UNDISTORT_IN_LUT
    Image* und_map_x = nullptr, * und_map_y = nullptr;
    {
        // undistort map alloc+compute: no file IO; we count it as "write" into RAM (two float maps)
        double bw = (double)WIDTH * (double)HEIGHT * sizeof(float) * 2.0;
        StageTimer T(STG_UNDMAP, 0, bw);
        generateUndistortMap(
            WIDTH, HEIGHT, fx, fy, cx, cy,  // distorted camera intrinsics
            fx_new, fy_new, cx_new, cy_new, // desired undistorted intrinsics
            K1, K2, K3, P1, P2,
            &und_map_x, &und_map_y);
    }
#endif

    if ((int)filenames.size() != NUM_CAMS) {
        printf("[ERROR] filenames.size() = %zu, but NUM_CAMS = %d\n", filenames.size(), NUM_CAMS);
        return 1;
    }

    // Crop in resized space (same as old pipeline)
    int crop_w = (int)(width_r * panorama_crop_width_scale);
    int crop_h = (int)(height_r * panorama_crop_height_scale);
    int crop_x0 = (width_r - crop_w) / 2;
    int crop_y0 = (height_r - crop_h) / 2;

    std::vector<Image*> frames(NUM_CAMS);
    std::vector<Image*> stabilized_frames;

    for (int cam_idx = 0; cam_idx < NUM_CAMS; ++cam_idx) {
        {
            // read one BMP (count raw bytes read)
            double br = (double)WIDTH * (double)HEIGHT * 3.0;
            StageTimer T(STG_IO, br, 0);
            frames[cam_idx] = readBMP(filenames[cam_idx].c_str());
        }
        if (!frames[cam_idx]) {
            printf("[ERROR] Failed to read image for camera %d: %s\n", cam_idx, filenames[cam_idx].c_str());
            continue;
        }
        printf("[INFO] Camera %d: Width = %d, Height = %d\n", cam_idx, frames[cam_idx]->width, frames[cam_idx]->height);

        float roll_cam = 0, pitch_cam = 0, yaw_cam = 0;
        convert_rpy_sensor_to_cam(
            roll_Sensor, pitch_Sensor,
            (cam_idx < (int)yaw_offset_cam.size() ? yaw_offset_cam[cam_idx] : 0.0f),
            (cam_idx < (int)pitch_offset_cam.size() ? pitch_offset_cam[cam_idx] : 0.0f),
            (cam_idx < (int)roll_offset_cam.size() ? roll_offset_cam[cam_idx] : 0.0f),
            &roll_cam, &pitch_cam, &yaw_cam);

        int y_shift = (int)(PITCH_TO_TRANSLATION_FACTOR * pitch_cam);

        // --- Build composite LUT for this camera (profiled) ---
        CompositeMap cmap;
        {
            // read cyl maps + undist maps (float), write LUT (MapEntry)
            double br = (double)width_r * (double)height_r * sizeof(float) * 2.0;
#if USE_UNDISTORT_IN_LUT
            br += (double)WIDTH * (double)HEIGHT * sizeof(float) * 2.0;
#endif
            double bw = (double)crop_w * (double)crop_h * sizeof(MapEntry);
            StageTimer T(STG_LUT, br, bw);
            buildCompositeMapForCam(
                map_x, map_y, width_r, height_r,
                crop_w, crop_h,
                crop_x0, crop_y0,
                roll_cam,        // +roll_cam (inverse mapping already handled)
                (float)y_shift,  // +y_shift
                resize_factor,
#if USE_UNDISTORT_IN_LUT
                und_map_x, und_map_y,
#endif
                cmap);
        }

        // --- Render with line-buffer (profiled) ---
        Image* stabilized = nullptr;
        {
            // read src image (RGB) + LUT; write cropped RGB
            double br = (double)WIDTH * (double)HEIGHT * 3.0 +
                (double)crop_w * (double)crop_h * sizeof(MapEntry);
            double bw = (double)crop_w * (double)crop_h * 3.0;
            StageTimer T(STG_RENDER, br, bw);
            stabilized = renderWithLineBuffer(frames[cam_idx], cmap);
        }
        stabilized_frames.push_back(stabilized);
    }

    // Blend all cams into panorama (profiled)
    Image* panorama = nullptr;
    {
        double br = 0;
        for (auto* s : stabilized_frames) if (s) br += (double)s->width * (double)s->height * 3.0;
        double bw = (double)3840 * (double)480 * 3.0;
        StageTimer T(STG_BLEND, br, bw);
        panorama = featherBlendFit(stabilized_frames, (int)(overlap_px * resize_factor), 3840, 480);
    }

    std::string result_path = SaveImageFilePath(dirCapture, L"panorama_linebuf");
    writeBMP(result_path.c_str(), panorama);
    printf("[Saved Panorama] %s\n", result_path.c_str());

    // print profile
    print_profile("PROFILING SUMMARY");

    // cleanup
    for (auto* f : frames) if (f) { free(f->data); free(f); }
    for (auto* s : stabilized_frames) { free(s->data); free(s); }
    free(panorama->data); free(panorama);
    free(map_x->data); free(map_x);
    free(map_y->data); free(map_y);
#if USE_UNDISTORT_IN_LUT
    free(und_map_x->data); free(und_map_x);
    free(und_map_y->data); free(und_map_y);
#endif
    return 0;
}
