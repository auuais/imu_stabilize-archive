// ===================== Profiling =====================
enum Stage : int { ST_IO = 0, ST_MAPS, ST_UNDISTMAP, ST_LUT, ST_RENDER, ST_BLEND, ST_RESIZE, ST_MISC, ST_COUNT };
static const char* STAGE_NAME[ST_COUNT] = { "I/O","Maps","UndistMap","LUT","Render","Blend","Resize","Misc" };

struct StageStats {
    std::atomic<uint64_t>  calls{ 0 };
    std::atomic<long double> ms_total{ 0.0L };
    std::atomic<uint64_t>  bytes_read{ 0 };
    std::atomic<uint64_t>  bytes_written{ 0 };
};
static StageStats G_STATS[ST_COUNT];
static thread_local Stage TL_CUR_STAGE = ST_MISC;

#if LIGHT_PROF
#define MEM_READ(bytes)   do{}while(0)
#define MEM_WRITE(bytes)  do{}while(0)
#else
#define MEM_READ(bytes)   (G_STATS[TL_CUR_STAGE].bytes_read   += (uint64_t)(bytes))
#define MEM_WRITE(bytes)  (G_STATS[TL_CUR_STAGE].bytes_written+= (uint64_t)(bytes))
#endif

struct StageTimer {
    Stage s; std::chrono::high_resolution_clock::time_point t0;
    StageTimer(Stage st) : s(st), t0(std::chrono::high_resolution_clock::now()) { TL_CUR_STAGE = s; G_STATS[s].calls++; }
    ~StageTimer() {
        auto t1 = std::chrono::high_resolution_clock::now();
        long double ms = std::chrono::duration<long double, std::milli>(t1 - t0).count();
        G_STATS[s].ms_total.store(G_STATS[s].ms_total.load() + ms);
        TL_CUR_STAGE = ST_MISC;
    }
};
static inline void print_stats_summary(const char* header) {
    printf("\r\n===== %s =====\r\n", header);
    for (int s = 0; s < ST_COUNT; ++s) {
        long double ms = G_STATS[s].ms_total.load();
        double sec = (double)ms / 1000.0;
        double rd_MB = (double)G_STATS[s].bytes_read.load() / (1024.0 * 1024.0);
        double wr_MB = (double)G_STATS[s].bytes_written.load() / (1024.0 * 1024.0);
        double rd_MBps = (sec > 0.0) ? (rd_MB / sec) : 0.0;
        double wr_MBps = (sec > 0.0) ? (wr_MB / sec) : 0.0;
        printf("Stage %-10s | calls:%4llu | time:%9.3Lf ms | R:%7.2f MB | W:%7.2f MB | R:%8.2f MB/s | W:%8.2f MB/s\r\n",
            STAGE_NAME[s],
            (unsigned long long)G_STATS[s].calls.load(),
            ms, rd_MB, wr_MB, rd_MBps, wr_MBps
        );
    }
    fflush(stdout);
}

// ===================== Tiny Thread Pool =====================
class TinyPool {
public:
    explicit TinyPool(unsigned n = std::thread::hardware_concurrency()) : stop(false) {
        n = n ? n : 1; for (unsigned i = 0; i < n; ++i) workers.emplace_back([this] { run(); });
    }
    ~TinyPool() { { std::unique_lock<std::mutex> lk(mx); stop = true; cv.notify_all(); } for (auto& t : workers) t.join(); }
    void submit(std::function<void()> fn) { { std::unique_lock<std::mutex> lk(mx); q.emplace(std::move(fn)); } cv.notify_one(); }
    void wait_idle() { std::unique_lock<std::mutex> lk(mx); done_cv.wait(lk, [&] { return q.empty() && active == 0; }); }
private:
    void run() {
        for (;;) {
            std::function<void()> job;
            {
                std::unique_lock<std::mutex> lk(mx);
                cv.wait(lk, [&] { return stop || !q.empty(); });
                if (stop && q.empty()) return;
                job = std::move(q.front()); q.pop(); ++active;
            }
            job();
            {
                std::unique_lock<std::mutex> lk(mx);
                --active; if (q.empty() && active == 0) done_cv.notify_all();
            }
        }
    }
    std::vector<std::thread> workers; std::mutex mx; std::condition_variable cv, done_cv; std::queue<std::function<void()>> q; std::atomic<bool> stop; int active = 0;
};

// ===================== Image (RGBX in-memory) =====================
struct Image {
    int width, height, channels; // channels=4 in memory (RGBX)
    int stride;                  // bytes per row (aligned)
    unsigned char* data;
};
static inline int align_up(int v, int a) { return (v + (a - 1)) & ~(a - 1); }

static Image* allocRGBX(int w, int h) {
    Image* img = (Image*)malloc(sizeof(Image));
    img->width = w; img->height = h; img->channels = 4;
    img->stride = align_up(w * 4, TILE_ROW_ALIGN_BYTES);
    img->data = (unsigned char*)_aligned_malloc((size_t)img->stride * img->height, TILE_ROW_ALIGN_BYTES);
    return img;
}
static void freeImage(Image* img) { if (!img) return; _aligned_free(img->data); free(img); }

// BMP I/O: read 24-bit; expand to RGBX aligned
static Image* readBMP_RGBX(const char* filename) {
    StageTimer _t(ST_IO);
    FILE* fp = fopen(filename, "rb"); if (!fp) return nullptr;
    BITMAPFILEHEADER bfh{}; BITMAPINFOHEADER bih{};
    fread(&bfh, sizeof(bfh), 1, fp); MEM_READ(sizeof(bfh));
    fread(&bih, sizeof(bih), 1, fp); MEM_READ(sizeof(bih));
    if (bfh.bfType != 0x4D42 || bih.biBitCount != 24 || bih.biCompression != BI_RGB) { fclose(fp); return nullptr; }
    const int w = bih.biWidth, h = std::abs(bih.biHeight);
    const int row_padded = ((w * 3 + 3) / 4) * 4;

    // Allocate RGBX
    Image* out = allocRGBX(w, h);
    fseek(fp, bfh.bfOffBits, SEEK_SET);

    // BMP is BGR; rows bottom-up if height positive (commonly negative means top-down)
    // We'll read top-down into RGBX.
    for (int y = 0; y < h; ++y) {
        unsigned char* row = (unsigned char*)_malloca(row_padded);
        size_t got = fread(row, 1, (size_t)row_padded, fp); MEM_READ(got);
        unsigned char* dst = &out->data[(size_t)(h - 1 - y) * out->stride];
        // Convert BGR -> RGBX, pad X=0
        for (int x = 0; x < w; ++x) {
            unsigned char b = row[x * 3 + 0];
            unsigned char g = row[x * 3 + 1];
            unsigned char r = row[x * 3 + 2];
            unsigned char* d = &dst[x * 4];
            d[0] = r; d[1] = g; d[2] = b; d[3] = 0;
        }
        _freea(row);
    }
    fclose(fp);
    return out;
}

// Write BMP from RGBX by stripping X and writing 24bpp
static int writeBMP_from_RGBX(const char* filename, const Image* img) {
    StageTimer _t(ST_IO);
    FILE* fp = fopen(filename, "wb"); if (!fp) return 0;
    const int w = img->width, h = img->height;
    const int row_rgb = w * 3;
    const int row_padded = ((row_rgb + 3) / 4) * 4;
    const int fileSize = 54 + row_padded * h;
    unsigned char bmpfileheader[14] = { 'B','M',0,0,0,0,0,0,0,0,54,0,0,0 };
    unsigned char bmpinfoheader[40] = { 0 };
    unsigned char pad[3] = { 0,0,0 };
    bmpfileheader[2] = (unsigned char)(fileSize);
    bmpfileheader[3] = (unsigned char)(fileSize >> 8);
    bmpfileheader[4] = (unsigned char)(fileSize >> 16);
    bmpfileheader[5] = (unsigned char)(fileSize >> 24);
    bmpinfoheader[0] = 40;
    bmpinfoheader[4] = (unsigned char)(w);
    bmpinfoheader[5] = (unsigned char)(w >> 8);
    bmpinfoheader[6] = (unsigned char)(w >> 16);
    bmpinfoheader[7] = (unsigned char)(w >> 24);
    bmpinfoheader[8] = (unsigned char)(h);
    bmpinfoheader[9] = (unsigned char)(h >> 8);
    bmpinfoheader[10] = (unsigned char)(h >> 16);
    bmpinfoheader[11] = (unsigned char)(h >> 24);
    bmpinfoheader[12] = 1; bmpinfoheader[14] = 24;

    fwrite(bmpfileheader, 1, 14, fp); MEM_WRITE(14);
    fwrite(bmpinfoheader, 1, 40, fp); MEM_WRITE(40);

    // write bottom-up
    for (int y = 0; y < h; ++y) {
        const unsigned char* src = &img->data[(size_t)(h - 1 - y) * img->stride];
        // strip X, write BGR order
        for (int x = 0; x < w; ++x) {
            const unsigned char* s = &src[x * 4];
            unsigned char bgr[3] = { s[2], s[1], s[0] };
            fwrite(bgr, 1, 3, fp); MEM_WRITE(3);
        }
        int padN = row_padded - row_rgb;
        if (padN > 0) { fwrite(pad, 1, padN, fp); MEM_WRITE(padN); }
    }
    fclose(fp); return 1;
}

// ===================== Cylindrical map Q12.4 =====================
static void loadCylMapsQ(std::vector<int16_t>& mapx, std::vector<int16_t>& mapy, int w, int h, const std::filesystem::path& exePath) {
    StageTimer _t(ST_MAPS);
    mapx.resize((size_t)w * h); mapy.resize((size_t)w * h);
    auto px = exePath.parent_path() / "map_x_q.bin";
    auto py = exePath.parent_path() / "map_y_q.bin";
    FILE* fxp = fopen(px.string().c_str(), "rb");
    FILE* fyp = fopen(py.string().c_str(), "rb");
    if (!fxp || !fyp) { if (fxp) fclose(fxp); if (fyp) fclose(fyp); throw std::runtime_error("Cyl maps not found"); }
    size_t gotx = fread(mapx.data(), sizeof(int16_t), (size_t)w * h, fxp); MEM_READ(gotx * sizeof(int16_t)); fclose(fxp);
    size_t goty = fread(mapy.data(), sizeof(int16_t), (size_t)w * h, fyp); MEM_READ(goty * sizeof(int16_t)); fclose(fyp);
    printf("[INFO] Cylindrical maps (Q12.4) loaded from disk.\r\n");
}
static void saveCylMapsQ(const std::vector<int16_t>& mapx, const std::vector<int16_t>& mapy, const std::filesystem::path& exePath) {
    StageTimer _t(ST_MAPS);
    auto px = exePath.parent_path() / "map_x_q.bin";
    auto py = exePath.parent_path() / "map_y_q.bin";
    FILE* fxp = fopen(px.string().c_str(), "wb");
    FILE* fyp = fopen(py.string().c_str(), "wb");
    fwrite(mapx.data(), sizeof(int16_t), mapx.size(), fxp); MEM_WRITE(mapx.size() * sizeof(int16_t)); fclose(fxp);
    fwrite(mapy.data(), sizeof(int16_t), mapy.size(), fyp); MEM_WRITE(mapy.size() * sizeof(int16_t)); fclose(fyp);
    printf("[INFO] Cylindrical maps (Q12.4) saved.\r\n");
}
static void generateCylMapsQ(std::vector<int16_t>& mapx, std::vector<int16_t>& mapy, int w, int h,
    float fov_h_deg, float fov_v_deg, float x_shift, const std::filesystem::path& exePath)
{
    StageTimer _t(ST_MAPS);
    mapx.resize((size_t)w * h); mapy.resize((size_t)w * h);
    float fov_h_rad = fov_h_deg * (float)M_PI / 180.0f, fov_v_rad = fov_v_deg * (float)M_PI / 180.0f;
    float fx_cyl = w / (2.0f * tanf(fov_h_rad / 2.0f));
    float fy_cyl = h / (2.0f * tanf(fov_v_rad / 2.0f));
    float cx_cyl = w / 2.0f, cy_cyl = h / 2.0f;

    for (int y = 0; y < h; ++y) {
        for (int x = 0; x < w; ++x) {
            float theta = (x - cx_cyl + x_shift) / fx_cyl;
            float hh = (y - cy_cyl) / fy_cyl;
            float xs = sinf(theta), ys = hh, zs = cosf(theta);
            float src_x = fx * resize_factor * (xs / zs) + cx * resize_factor;
            float src_y = fy * resize_factor * (ys / zs) + cy * resize_factor;
            int32_t qx = (int32_t)lroundf(src_x * MAP_SCALE);
            int32_t qy = (int32_t)lroundf(src_y * MAP_SCALE);
            if (qx < INT16_MIN) qx = INT16_MIN; if (qx > INT16_MAX) qx = INT16_MAX;
            if (qy < INT16_MIN) qy = INT16_MIN; if (qy > INT16_MAX) qy = INT16_MAX;
            mapx[(size_t)y * w + x] = (int16_t)qx; MEM_WRITE(sizeof(int16_t));
            mapy[(size_t)y * w + x] = (int16_t)qy; MEM_WRITE(sizeof(int16_t));
        }
    }
    saveCylMapsQ(mapx, mapy, exePath);
}

// ===================== Angle LUT (−5..+5 step 0.25) Q1.15 =====================
struct AngleLUTEntry { int16_t c; int16_t s; };
static std::vector<AngleLUTEntry> ANG_LUT;
static inline void build_angle_lut() {
    if (!ANG_LUT.empty()) return;
    const float minDeg = -5.0f, maxDeg = 5.0f, step = 0.25f;
    int N = (int)std::lround((maxDeg - minDeg) / step) + 1;
    ANG_LUT.resize(N);
    for (int i = 0; i < N; ++i) {
        float deg = minDeg + i * step;
        float rad = deg * (float)M_PI / 180.0f;
        int16_t c = (int16_t)std::lroundf(cosf(rad) * 32767.0f);  // Q1.15
        int16_t s = (int16_t)std::lroundf(sinf(rad) * 32767.0f);  // Q1.15
        ANG_LUT[i] = { c, s };
    }
}
static inline void get_cos_sin_q15(float deg, int16_t* cos_q15, int16_t* sin_q15) {
    // nearest index
    float idxf = (deg + 5.0f) / 0.25f;
    int idx = (int)std::lround(idxf);
    if (idx < 0) idx = 0; if (idx >= (int)ANG_LUT.size()) idx = (int)ANG_LUT.size() - 1;
    *cos_q15 = ANG_LUT[idx].c; *sin_q15 = ANG_LUT[idx].s;
}

// ===================== LUT: packed 6 bytes (x0,y0 12b + wx8, wy8 + pad) =====================
#pragma pack(push,1)
struct MapEntryPacked48 {
    // [ 11:0 x0 ][ 11:0 y0 ][ 7:0 wx8 ][ 7:0 wy8 ][ 7:0 pad ]
    uint16_t xy0_hi;  // bits: [15:4] x0_hi(12), [3:0] y0_hi(4)
    uint8_t  y0_lo;   // bits: [7:0] y0_lo
    uint8_t  wx8;
    uint8_t  wy8;
    uint8_t  pad;
};
#pragma pack(pop)
static inline int clampi(int v, int lo, int hi) { if (v < lo) v = lo; if (v > hi) v = hi; return v; }

// Build composite LUT (packed, Q8.8 weights), uses angle LUT for sin/cos
static void buildCompositeLUT_Packed(
    std::vector<MapEntryPacked48>& lut,
    const std::vector<int16_t>& mapx_q,
    const std::vector<int16_t>& mapy_q,
    int in_w, int in_h,
    int resized_w, int resized_h,
    int crop_w, int crop_h,
    int crop_x0, int crop_y0,
    float roll_deg, int y_shift_px,
    float resize_factor_local)
{
    StageTimer _t(ST_LUT);
    lut.resize((size_t)crop_w * crop_h);

    const int32_t cxr_q = (resized_w << 15);
    const int32_t cyr_q = (resized_h << 15);

    int16_t cos15 = 0, sin15 = 0; get_cos_sin_q15(roll_deg, &cos15, &sin15); // Q1.15
    const int32_t yshift_q = ((int32_t)y_shift_px) << WFIX_SHIFT;
    const int32_t inv_resize_q = (int32_t)llroundf((1.0f / resize_factor_local) * (1 << WFIX_SHIFT));
    const int32_t max_x_q = ((in_w - 2) << WFIX_SHIFT);
    const int32_t max_y_q = ((in_h - 2) << WFIX_SHIFT);

    for (int y = 0; y < crop_h; ++y) {
        const int tgt_y = crop_y0 + y;
        const size_t base = (size_t)tgt_y * resized_w + crop_x0;
        const int16_t* __restrict mx = &mapx_q[base];
        const int16_t* __restrict my = &mapy_q[base];
        MapEntryPacked48* __restrict me = &lut[(size_t)y * crop_w];

        for (int x = 0; x < crop_w; ++x) {
            // src in Q12.4 -> Q16.16
            const int32_t srcx_q16 = ((int32_t)mx[x]) << (WFIX_SHIFT - MAP_SHIFT);
            const int32_t srcy_q16 = ((int32_t)my[x]) << (WFIX_SHIFT - MAP_SHIFT);

            const int32_t dx_q = srcx_q16 - cxr_q;
            const int32_t dy_q = (srcy_q16 - cyr_q) - yshift_q;

            // rotate with Q1.15 -> (cos*dx + sin*dy) >> 15 => Q16.16
            int32_t ax_q = (int32_t)(((int64_t)cos15 * dx_q + (int64_t)sin15 * dy_q) >> 15) + cxr_q;
            int32_t ay_q = (int32_t)((-(int64_t)sin15 * dx_q + (int64_t)cos15 * dy_q) >> 15) + cyr_q;

            // inverse resize (Q16.16)
            int32_t ox_q = (int32_t)(((int64_t)ax_q * inv_resize_q) >> WFIX_SHIFT);
            int32_t oy_q = (int32_t)(((int64_t)ay_q * inv_resize_q) >> WFIX_SHIFT);

            ox_q = clampi(ox_q, 0, max_x_q);
            oy_q = clampi(oy_q, 0, max_y_q);

            const int x0 = ox_q >> 16;
            const int y0 = oy_q >> 16;
            const uint8_t wx8 = (uint8_t)((ox_q >> 8) & 0xFF);
            const uint8_t wy8 = (uint8_t)((oy_q >> 8) & 0xFF);

            // pack 12-bit x0, 12-bit y0
            MapEntryPacked48 e;
            const uint16_t x0_12 = (uint16_t)(x0 & 0x0FFF);
            const uint16_t y0_12 = (uint16_t)(y0 & 0x0FFF);
            e.xy0_hi = (uint16_t)((x0_12 << 4) | (y0_12 >> 8)); // [x0_hi(12)][y0_hi(4)]
            e.y0_lo = (uint8_t)(y0_12 & 0xFF);
            e.wx8 = wx8; e.wy8 = wy8; e.pad = 0;
            me[x] = e; MEM_WRITE(sizeof(MapEntryPacked48));
        }
    }
}

// Unpack helpers for render
static inline void unpack_xy_w(const MapEntryPacked48& e, int& x0, int& y0, uint8_t& wx8, uint8_t& wy8) {
    uint16_t xyh = e.xy0_hi;
    uint16_t x0_12 = (uint16_t)((xyh >> 4) & 0x0FFF);
    uint16_t y0_hi4 = (uint16_t)(xyh & 0x000F);
    uint16_t y0_12 = (uint16_t)((y0_hi4 << 8) | e.y0_lo);
    x0 = (int)x0_12; y0 = (int)y0_12;
    wx8 = e.wx8; wy8 = e.wy8;
}

// ===================== Renderer (tile, separable bilinear, RGBX) =====================
static void renderFromLUT_Tiled_RGBX(
    const Image* src,
    const std::vector<MapEntryPacked48>& lut,
    int out_w, int out_h,
    Image* out)
{
    StageTimer _t(ST_RENDER);
    out->width = out_w; out->height = out_h; out->channels = 4; // RGBX
    out->stride = align_up(out_w * 4, TILE_ROW_ALIGN_BYTES);
    out->data = (unsigned char*)_aligned_malloc((size_t)out->stride * out->height, TILE_ROW_ALIGN_BYTES);

    const int s_stride = src->stride;

    for (int ty = 0; ty < out_h; ty += TILE_H) {
        int h_block = (ty + TILE_H <= out_h) ? TILE_H : (out_h - ty);
        for (int tx = 0; tx < out_w; tx += TILE_W) {
            int w_block = (tx + TILE_W <= out_w) ? TILE_W : (out_w - tx);

            // Process tile (separable bilinear). HW equivalent: 2 line buffers.
            for (int y = 0; y < h_block; ++y) {
                int oy = ty + y;
                const MapEntryPacked48* rowLUT = &lut[(size_t)oy * out_w + tx];
                unsigned char* dst = &out->data[(size_t)oy * out->stride + tx * 4];

                for (int x = 0; x < w_block; ++x) {
                    int x0, y0; uint8_t wx8, wy8;
                    unpack_xy_w(rowLUT[x], x0, y0, wx8, wy8);

                    // Bounds safe due to LUT clamp
                    const unsigned char* p0 = &src->data[(size_t)y0 * s_stride + x0 * 4];
                    const unsigned char* p1 = p0 + 4;                 // (x0+1, y0)
                    const unsigned char* p2 = p0 + s_stride;          // (x0,   y0+1)
                    const unsigned char* p3 = p2 + 4;                 // (x0+1, y0+1)

                    // Weights (separable Q8.8)
                    const int W_x0 = 256 - (int)wx8;
                    const int W_x1 = (int)wx8;
                    const int W_y0 = 256 - (int)wy8;
                    const int W_y1 = (int)wy8;

                    // Horizontal lerp (h0, h1) per channel (RGB only)
                    // h = a + ((b - a) * wx8) >> 8
                    int r0 = p0[0] + (((int)p1[0] - (int)p0[0]) * W_x1 >> 8);
                    int g0 = p0[1] + (((int)p1[1] - (int)p0[1]) * W_x1 >> 8);
                    int b0 = p0[2] + (((int)p1[2] - (int)p0[2]) * W_x1 >> 8);

                    int r1 = p2[0] + (((int)p3[0] - (int)p2[0]) * W_x1 >> 8);
                    int g1 = p2[1] + (((int)p3[1] - (int)p2[1]) * W_x1 >> 8);
                    int b1 = p2[2] + (((int)p3[2] - (int)p2[2]) * W_x1 >> 8);

                    // Vertical lerp: v = h0 + ((h1 - h0) * wy8) >> 8
                    unsigned char* d = &dst[x * 4];
                    d[0] = (unsigned char)(r0 + (((r1 - r0) * W_y1) >> 8));
                    d[1] = (unsigned char)(g0 + (((g1 - g0) * W_y1) >> 8));
                    d[2] = (unsigned char)(b0 + (((b1 - b0) * W_y1) >> 8));
                    d[3] = 0;
                }
            }
        }
    }
}

// ===================== Feather Blend + Resize (RGBX) =====================
static Image* featherBlendFitMT_RGBX(const std::vector<Image*>& frames, int overlap, int target_w, int target_h) {
    StageTimer _t(ST_BLEND);
    if (frames.empty()) return nullptr;
    const int h = frames[0]->height, c = 4;
    int total_w = frames[0]->width;
    for (size_t i = 1; i < frames.size(); ++i) total_w += (frames[i]->width - overlap);

    Image* blended = allocRGBX(total_w, h);

    // Precompute alpha ramp Q0.16
    std::vector<uint16_t> alpha(overlap);
    for (int x = 0; x < overlap; ++x) alpha[x] = (uint16_t)(((uint64_t)x << 16) / (overlap ? overlap : 1));

    // Row-parallel blend
    TinyPool pool(std::max(1u, std::thread::hardware_concurrency()));
    auto worker = [&](int y0, int y1) {
        for (int y = y0; y < y1; ++y) {
            unsigned char* dstRow = &blended->data[(size_t)y * blended->stride];

            // seed with first frame row
            const Image* img0 = frames[0];
            memcpy(dstRow, &img0->data[(size_t)y * img0->stride], (size_t)img0->width * 4);
            int curx = img0->width;

            for (size_t i = 1; i < frames.size(); ++i) {
                const Image* img = frames[i];
                const unsigned char* srcRow = &img->data[(size_t)y * img->stride];

                int ov = overlap; if (ov > img->width) ov = img->width; if (ov > curx) ov = curx;
                // blend overlap (RGB channels)
                for (int x = 0; x < ov; ++x) {
                    unsigned a = alpha[x];
                    size_t dp = (size_t)(curx - ov + x) * 4, sp = (size_t)x * 4;
                    for (int ch = 0; ch < 3; ++ch) {
                        unsigned prev = dstRow[dp + ch];
                        unsigned curr = srcRow[sp + ch];
                        dstRow[dp + ch] = (unsigned char)((prev * (0x10000 - a) + curr * a) >> 16);
                    }
                }
                // copy tail
                int tail = img->width - ov;
                if (tail > 0) memcpy(&dstRow[(size_t)curx * 4], &srcRow[(size_t)ov * 4], (size_t)tail * 4);
                curx += tail;
            }
        }
        };
    const int chunk = 128;
    for (int y0 = 0; y0 < h; y0 += chunk) pool.submit([&, y0] { worker(y0, std::min(h, y0 + chunk)); });
    pool.wait_idle();

    // === Resize to target_w using row-parallel bilinear (Q8.8 separable) ===
    StageTimer _r(ST_RESIZE);
    float scale = (float)target_w / (float)blended->width;
    int new_w = target_w;
    int new_h = (int)lroundf(blended->height * scale);

    Image* resized = allocRGBX(new_w, new_h);

    const int32_t inv_scale_q = (int32_t)llroundf((1.0f / scale) * (1 << 16));

    // Precompute sx/wx for all x
    std::vector<int> sx_arr(new_w);
    std::vector<uint8_t> wx8_arr(new_w);
    for (int x = 0; x < new_w; ++x) {
        int32_t sx_q = ((int32_t)x << 16);
        sx_q = (int32_t)(((int64_t)sx_q * inv_scale_q) >> 16);
        int sx = clampi(sx_q >> 16, 0, blended->width - 2);
        sx_arr[x] = sx;
        wx8_arr[x] = (uint8_t)((sx_q >> 8) & 0xFF);
    }

    TinyPool pool2(std::max(1u, std::thread::hardware_concurrency()));
    auto resize_worker = [&](int y0, int y1) {
        for (int y = y0; y < y1; ++y) {
            int32_t sy_q = ((int32_t)y << 16);
            sy_q = (int32_t)(((int64_t)sy_q * inv_scale_q) >> 16);
            int sy = clampi(sy_q >> 16, 0, blended->height - 2);
            uint8_t wy8 = (uint8_t)((sy_q >> 8) & 0xFF);

            const unsigned char* row0 = &blended->data[(size_t)sy * blended->stride];
            const unsigned char* row1 = row0 + blended->stride;

            unsigned char* drow = &resized->data[(size_t)y * resized->stride];

            for (int x = 0; x < new_w; ++x) {
                int sx = sx_arr[x]; uint8_t wx8 = wx8_arr[x];
                // h0, h1 per channel
                const unsigned char* p0 = &row0[sx * 4];
                const unsigned char* p1 = p0 + 4;
                const unsigned char* p2 = &row1[sx * 4];
                const unsigned char* p3 = p2 + 4;

                int r0 = p0[0] + (((int)p1[0] - (int)p0[0]) * wx8 >> 8);
                int g0 = p0[1] + (((int)p1[1] - (int)p0[1]) * wx8 >> 8);
                int b0 = p0[2] + (((int)p1[2] - (int)p0[2]) * wx8 >> 8);

                int r1 = p2[0] + (((int)p3[0] - (int)p2[0]) * wx8 >> 8);
                int g1 = p2[1] + (((int)p3[1] - (int)p2[1]) * wx8 >> 8);
                int b1 = p2[2] + (((int)p3[2] - (int)p2[2]) * wx8 >> 8);

                unsigned char* d = &drow[x * 4];
                d[0] = (unsigned char)(r0 + (((r1 - r0) * wy8) >> 8));
                d[1] = (unsigned char)(g0 + (((g1 - g0) * wy8) >> 8));
                d[2] = (unsigned char)(b0 + (((b1 - b0) * wy8) >> 8));
                d[3] = 0;
            }
        }
        };
    const int chunk2 = 128;
    for (int y0 = 0; y0 < new_h; y0 += chunk2) pool2.submit([&, y0] { resize_worker(y0, std::min(new_h, y0 + chunk2)); });
    pool2.wait_idle();

    // Fit to final target (center crop/pad vertically)
    Image* finalOut = allocRGBX(target_w, target_h);
    memset(finalOut->data, 0, (size_t)finalOut->stride * finalOut->height);
    int yoff = (target_h - new_h) / 2; if (yoff < 0) yoff = 0; int copy_h = (new_h > target_h) ? target_h : new_h;
    for (int y = 0; y < copy_h; ++y) {
        if (y + yoff < 0 || y + yoff >= target_h) continue;
        memcpy(&finalOut->data[(size_t)(y + yoff) * finalOut->stride],
            &resized->data[(size_t)y * resized->stride],
            (size_t)new_w * 4);
    }

    freeImage(resized);
    freeImage(blended);
    return finalOut;
}

// ===================== Sensor to camera =====================
static void convert_rpy_sensor_to_cam(double roll_sys1, double pitch_sys1,
    double yaw_offset, double pitch_offset, double roll_offset,
    float* roll_deg_out, float* pitch_deg_out, float* yaw_deg_out)
{
    StageTimer _t(ST_MISC);
    double rr = roll_sys1 * M_PI / 180.0, pp = pitch_sys1 * M_PI / 180.0;
    double ror = roll_offset * M_PI / 180.0, por = pitch_offset * M_PI / 180.0, yor = yaw_offset * M_PI / 180.0;

    double Rx[3][3] = { {1,0,0},{0,cos(rr),-sin(rr)},{0,sin(rr),cos(rr)} };
    double Ry[3][3] = { {cos(pp),0,sin(pp)},{0,1,0},{-sin(pp),0,cos(pp)} };
    double R_sys1[3][3];
    for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j)
        R_sys1[i][j] = Rx[i][0] * Ry[0][j] + Rx[i][1] * Ry[1][j] + Rx[i][2] * Ry[2][j];

    double Rr[3][3] = { {1,0,0},{0,cos(ror),-sin(ror)},{0,sin(ror),cos(ror)} };
    double Rp[3][3] = { {cos(por),0,sin(por)},{0,1,0},{-sin(por),0,cos(por)} };
    double Rz[3][3] = { {cos(yor),-sin(yor),0},{sin(yor),cos(yor),0},{0,0,1} };
    double Rt[3][3];
    for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j)
        Rt[i][j] = Rz[i][0] * Rp[0][j] + Rz[i][1] * Rp[1][j] + Rz[i][2] * Rp[2][j];

    double Roff[3][3];
    for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j)
        Roff[i][j] = Rt[i][0] * Rr[0][j] + Rt[i][1] * Rr[1][j] + Rt[i][2] * Rr[2][j];

    double R[3][3];
    for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j)
        R[i][j] = R_sys1[i][0] * Roff[0][j] + R_sys1[i][1] * Roff[1][j] + R_sys1[i][2] * Roff[2][j];

    double pitchX = asin(-R[2][0]); double cy = cos(pitchX);
    double rollX, yawX;
    if (fabs(cy) > 1e-6) { rollX = atan2(R[2][1], R[2][2]); yawX = atan2(R[1][0], R[0][0]); }
    else { rollX = 0; yawX = atan2(-R[0][1], R[1][1]); }

    *roll_deg_out = (float)(rollX * 180.0 / M_PI);
    *pitch_deg_out = (float)(pitchX * 180.0 / M_PI);
    *yaw_deg_out = (float)(yawX * 180.0 / M_PI);
}

// ===================== single-frame runner =====================
static double process_once(
    const std::filesystem::path& exePath,
    const std::vector<std::string>& filenames,
    const std::vector<int16_t>& mapx_q,
    const std::vector<int16_t>& mapy_q,
    int resized_w, int resized_h,
    int crop_w, int crop_h, int crop_x0, int crop_y0,
    const std::vector<float>& yaw_off,
    const std::vector<float>& pit_off,
    const std::vector<float>& rol_off,
    float PITCH_TO_TRANSLATION_FACTOR,
    float resize_factor_local,
    int overlap_px_resized)
{
    auto t0 = std::chrono::high_resolution_clock::now();

    // read six frames fresh (as requested)
    std::vector<Image*> frames(NUM_CAMS, nullptr);
    {
        StageTimer _t(ST_IO);
        for (int i = 0; i < NUM_CAMS; ++i) {
            frames[i] = readBMP_RGBX(filenames[i].c_str());
            if (!frames[i]) { printf("[ERROR] read fail cam %d\r\n", i); std::exit(1); }
        }
    }

    // per-camera stabilize
    std::vector<Image*> stabilized(NUM_CAMS, nullptr);
#if ENABLE_MT
    {
        TinyPool pool(std::max(1u, std::thread::hardware_concurrency()));
        for (int cam = 0; cam < NUM_CAMS; ++cam) {
            pool.submit([&, cam] {
                float roll_cam = 0, pitch_cam = 0, yaw_cam = 0;
                convert_rpy_sensor_to_cam(roll_Sensor, pitch_Sensor,
                    yaw_off[cam], pit_off[cam], rol_off[cam],
                    &roll_cam, &pitch_cam, &yaw_cam);

                int y_shift_px = (int)lroundf(PITCH_TO_TRANSLATION_FACTOR * pitch_cam);

                std::vector<MapEntryPacked48> lut; lut.reserve((size_t)crop_w * crop_h);
                buildCompositeLUT_Packed(lut, mapx_q, mapy_q,
                    frames[cam]->width, frames[cam]->height,
                    resized_w, resized_h,
                    crop_w, crop_h, crop_x0, crop_y0,
                    -roll_cam, -y_shift_px, resize_factor_local);

                Image tmp{};
                renderFromLUT_Tiled_RGBX(frames[cam], lut, crop_w, crop_h, &tmp);

                Image* out = allocRGBX(crop_w, crop_h);
                for (int y = 0; y < tmp.height; ++y) {
                    memcpy(&out->data[(size_t)y * out->stride],
                        &tmp.data[(size_t)y * tmp.stride],
                        (size_t)crop_w * 4);
                }
                out->channels = tmp.channels; out->stride = align_up(crop_w * 4, TILE_ROW_ALIGN_BYTES);
                _aligned_free(tmp.data);
                stabilized[cam] = out;
                });
        }
        pool.wait_idle();
    }
#else
    for (int cam = 0; cam < NUM_CAMS; ++cam) { /* same as above body */ }
#endif

    // blend + resize
    Image* panorama = featherBlendFitMT_RGBX(stabilized, overlap_px_resized, 3840, 480);

#if SAVE_EACH_FRAME
    {
        std::wstring captureDirW = CAPTURE_DIR_W; CreateDirectoryW(captureDirW.c_str(), NULL);
        std::string savePath = SaveImageFilePath(captureDirW, L"panorama_v4_rt");
        Image writeView = *panorama;
        writeBMP_from_RGBX(savePath.c_str(), &writeView);
    }
#endif

    // free images
    for (auto* f : frames)     if (f) freeImage(f);
    for (auto* s : stabilized) if (s) freeImage(s);
    freeImage(panorama);

    auto t1 = std::chrono::high_resolution_clock::now();
    return std::chrono::duration<double, std::milli>(t1 - t0).count(); // ms
}

// Run a frame using frames already in RAM (no disk I/O)
static double process_from_cached(
    const std::vector<Image*>& frames,          // RGBX, same geometry each iter
    const std::vector<int16_t>& mapx_q,
    const std::vector<int16_t>& mapy_q,
    int resized_w, int resized_h,
    int crop_w, int crop_h, int crop_x0, int crop_y0,
    const std::vector<float>& yaw_off,
    const std::vector<float>& pit_off,
    const std::vector<float>& rol_off,
    float PITCH_TO_TRANSLATION_FACTOR,
    float resize_factor_local,
    int overlap_px_resized)
{
    auto t0 = std::chrono::high_resolution_clock::now();

    // per-camera stabilize (same as in process_once, but no reads)
    std::vector<Image*> stabilized(NUM_CAMS, nullptr);
#if ENABLE_MT
    {
        TinyPool pool(std::max(1u, std::thread::hardware_concurrency()));
        for (int cam = 0; cam < NUM_CAMS; ++cam) {
            pool.submit([&, cam] {
                float roll_cam = 0, pitch_cam = 0, yaw_cam = 0;
                convert_rpy_sensor_to_cam(roll_Sensor, pitch_Sensor,
                    yaw_off[cam], pit_off[cam], rol_off[cam],
                    &roll_cam, &pitch_cam, &yaw_cam);

                int y_shift_px = (int)lroundf(PITCH_TO_TRANSLATION_FACTOR * pitch_cam);

                std::vector<MapEntryPacked48> lut; lut.reserve((size_t)crop_w * crop_h);
                buildCompositeLUT_Packed(lut, mapx_q, mapy_q,
                    frames[cam]->width, frames[cam]->height,
                    resized_w, resized_h,
                    crop_w, crop_h, crop_x0, crop_y0,
                    -roll_cam, -y_shift_px, resize_factor_local);

                Image tmp{};
                renderFromLUT_Tiled_RGBX(frames[cam], lut, crop_w, crop_h, &tmp);

                Image* out = allocRGBX(crop_w, crop_h);
                for (int y = 0; y < tmp.height; ++y) {
                    memcpy(&out->data[(size_t)y * out->stride],
                        &tmp.data[(size_t)y * tmp.stride],
                        (size_t)crop_w * 4);
                }
                out->channels = tmp.channels; out->stride = align_up(crop_w * 4, TILE_ROW_ALIGN_BYTES);
                _aligned_free(tmp.data);
                stabilized[cam] = out;
                });
        }
        pool.wait_idle();
    }
#else
    // single-thread: same body
#endif

    Image* panorama = featherBlendFitMT_RGBX(stabilized, overlap_px_resized, 3840, 480);

#if SAVE_PANORAMA_EACH_RUN
    {
        std::wstring captureDirW = CAPTURE_DIR_W; CreateDirectoryW(captureDirW.c_str(), NULL);
        std::string savePath = SaveImageFilePath(captureDirW, L"panorama_v4_rt");
        writeBMP_from_RGBX(savePath.c_str(), panorama);
        printf("[Saved Panorama] %s\r\n", savePath.c_str());
    }
#endif

    for (auto* s : stabilized) if (s) freeImage(s);
    freeImage(panorama);

    auto t1 = std::chrono::high_resolution_clock::now();
    return std::chrono::duration<double, std::milli>(t1 - t0).count();
}


// ===================== main (replaces your current main) =====================
int main() {
    build_angle_lut();

    // exe and INIs
    wchar_t wpath[MAX_PATH] = L""; GetModuleFileNameW(NULL, wpath, MAX_PATH);
    std::filesystem::path exePath(wpath);
    std::filesystem::path imuIniPath = exePath;        imuIniPath.replace_extension(L".ini");
    std::filesystem::path paramsIniPath = exePath;     paramsIniPath.replace_filename(L"parameters.ini");

    // params
    INIManager paramIni(paramsIniPath);
    WIDTH = paramIni.ReadInteger(L"image", L"width", WIDTH);
    HEIGHT = paramIni.ReadInteger(L"image", L"height", HEIGHT);
    NUM_CAMS = paramIni.ReadInteger(L"image", L"num_cams", NUM_CAMS);

    fx = paramIni.ReadFloat(L"calibration", L"fx", fx);
    fy = paramIni.ReadFloat(L"calibration", L"fy", fy);
    float cx_ratio = paramIni.ReadFloat(L"calibration", L"cx_ratio", 0.5f);
    float cy_ratio = paramIni.ReadFloat(L"calibration", L"cy_ratio", 0.5f);
    cx = WIDTH * cx_ratio; cy = HEIGHT * cy_ratio;

    cylindrical_H_fov_deg = paramIni.ReadFloat(L"projection", L"cylindrical_H_fov_deg", cylindrical_H_fov_deg);
    cylindrical_V_fov_deg = paramIni.ReadFloat(L"projection", L"cylindrical_V_fov_deg", cylindrical_V_fov_deg);
    panorama_crop_height_scale = paramIni.ReadFloat(L"cropping", L"panorama_crop_height_scale", panorama_crop_height_scale);
    panorama_crop_width_scale = paramIni.ReadFloat(L"cropping", L"panorama_crop_width_scale", panorama_crop_width_scale);
    resize_factor = paramIni.ReadFloat(L"resize", L"resize_factor ", resize_factor);
    PITCH_TO_TRANSLATION_FACTOR = paramIni.ReadFloat(L"transform", L"pitch_to_translation_factor", PITCH_TO_TRANSLATION_FACTOR) * resize_factor;
    YAW_TO_TRANSLATION_FACTOR = paramIni.ReadFloat(L"transform", L"yaw_to_translation_factor", YAW_TO_TRANSLATION_FACTOR);
    overlap_px = paramIni.ReadInteger(L"blend", L"overlap_px ", overlap_px);

    // IMU ini
    INIManager imuIni(imuIniPath);

    // file list
    std::vector<std::string> filenames; filenames.reserve(NUM_CAMS);
    for (int i = 0; i < NUM_CAMS; ++i) {
        std::wstring key = L"file_" + std::to_wstring(i);
        filenames.push_back(WStringToUtf8(imuIni.ReadString(L"images", key, L"")));
    }
    std::vector<float> yaw_off = imuIni.ReadFloatArray(L"offset", L"yaw");
    std::vector<float> pit_off = imuIni.ReadFloatArray(L"offset", L"pitch");
    std::vector<float> rol_off = imuIni.ReadFloatArray(L"offset", L"roll");

    auto clamp5 = [&](float v) { return v > 5.0f ? 5.0f : (v < -5.0f ? -5.0f : v); };
    bool StabilizationIni = imuIni.ReadBoolean(L"Sensor_Orientation", L"stabilization", true);
    if (StabilizationIni) {
        roll_Sensor = clamp5(imuIni.ReadFloat(L"Sensor_Orientation", L"roll", 0.0f));
        pitch_Sensor = clamp5(imuIni.ReadFloat(L"Sensor_Orientation", L"pitch", 0.0f));
        yaw_Sensor = clamp5(imuIni.ReadFloat(L"Sensor_Orientation", L"yaw", 0.0f));
    }
    else { roll_Sensor = pitch_Sensor = yaw_Sensor = 0.0f; }

    // geometry
    const int resized_w = (int)lroundf(WIDTH * resize_factor);
    const int resized_h = (int)lroundf(HEIGHT * resize_factor);
    const int crop_w = (int)lroundf(resized_w * panorama_crop_width_scale);
    const int crop_h = (int)lroundf(resized_h * panorama_crop_height_scale);
    const int crop_x0 = (resized_w - crop_w) / 2;
    const int crop_y0 = (resized_h - crop_h) / 2;
    const int overlap_resized = (int)lroundf(overlap_px * resize_factor);

    // maps (once)
    std::vector<int16_t> mapx_q, mapy_q;
    try { loadCylMapsQ(mapx_q, mapy_q, resized_w, resized_h, exePath); }
    catch (...) { generateCylMapsQ(mapx_q, mapy_q, resized_w, resized_h, cylindrical_H_fov_deg, cylindrical_V_fov_deg, 0.0f, exePath); }

#if RUN_REALTIME_LOOP
    logln("........Processing (V4-RT-Loop, cached RAM frames, steady-state)........");

    // Warm-up: load once from disk to RAM (simulate DDR resident images)
    std::vector<Image*> cached(NUM_CAMS, nullptr);
    {
        StageTimer _t(ST_IO);
        for (int i = 0; i < NUM_CAMS; ++i) {
            cached[i] = readBMP_RGBX(filenames[i].c_str());
            if (!cached[i]) { printf("[ERROR] read fail cam %d\r\n", i); return 1; }
        }
    }

    // First iteration (warm pipeline), do not include in average
    for (int s = 0; s < ST_COUNT; ++s) { G_STATS[s].calls = 0; G_STATS[s].ms_total = 0.0L; }
    double warm_ms = process_from_cached(cached, mapx_q, mapy_q,
        resized_w, resized_h,
        crop_w, crop_h, crop_x0, crop_y0,
        yaw_off, pit_off, rol_off,
        PITCH_TO_TRANSLATION_FACTOR, resize_factor,
        overlap_resized);
    printf("[WARMUP] total = %.3f ms (%.2f FPS)\r\n", warm_ms, (warm_ms > 0.0) ? 1000.0 / warm_ms : 0.0);

    // Steady-state iterations
    double sum_ms = 0.0;
    for (int it = 0; it < ITERATIONS; ++it) {
        for (int s = 0; s < ST_COUNT; ++s) { G_STATS[s].calls = 0; G_STATS[s].ms_total = 0.0L; }
        double ms = process_from_cached(cached, mapx_q, mapy_q,
            resized_w, resized_h,
            crop_w, crop_h, crop_x0, crop_y0,
            yaw_off, pit_off, rol_off,
            PITCH_TO_TRANSLATION_FACTOR, resize_factor,
            overlap_resized);
        sum_ms += ms;
        double fps = (ms > 0.0) ? (1000.0 / ms) : 0.0;
        printf("[ITER %2d] total = %.3f ms (%.2f FPS)\r\n", it, ms, fps);
        print_stats_summary("PROFILING SUMMARY (per-iter, cached)");
}

    // Average over steady-state only
    double avg_ms = sum_ms / ITERATIONS;
    double avg_fps = (avg_ms > 0.0) ? (1000.0 / avg_ms) : 0.0;
    printf("\r\n===== STEADY-STATE over %d iters (disk I/O excluded) =====\r\n", ITERATIONS);
    printf("[TIME] Avg total: %.3f ms  (%.2f FPS)\r\n", avg_ms, avg_fps);

    // Cleanup cached frames
    for (auto* f : cached) if (f) freeImage(f);


#else
    // single frame run (previous behavior)
    logln("........Processing (V3-FPGA-Twin, RGBX, packed LUT, tiles)........");
    double ms = process_once(exePath, filenames, mapx_q, mapy_q,
        resized_w, resized_h, crop_w, crop_h, crop_x0, crop_y0,
        yaw_off, pit_off, rol_off,
        PITCH_TO_TRANSLATION_FACTOR, resize_factor,
        overlap_resized);
    printf("[TIME] Total: %.3f ms (%.2f FPS)\r\n", ms, (ms > 0.0) ? 1000.0 / ms : 0.0);
#endif
    return 0;
}
