// ===================== Profiling =====================
enum Stage : int { ST_IO = 0, ST_MAPS, ST_UNDISTMAP, ST_LUT, ST_RENDER, ST_BLEND, ST_RESIZE, ST_MISC, ST_COUNT };
static const char* STAGE_NAME[ST_COUNT] = { "I/O","Maps","UndistMap","LUT","Render","Blend","Resize","Misc" };

struct StageStats {
    std::atomic<uint64_t> calls{ 0 };
    std::atomic<long double> ms_total{ 0.0L };
    std::atomic<uint64_t> bytes_read{ 0 };
    std::atomic<uint64_t> bytes_written{ 0 };
};
static StageStats G_STATS[ST_COUNT];
static thread_local Stage TL_CUR_STAGE = ST_MISC;

#if LIGHT_PROF
#define MEM_READ(bytes)   do{}while(0)
#define MEM_WRITE(bytes)  do{}while(0)
#else
#define MEM_READ(bytes)   (G_STATS[TL_CUR_STAGE].bytes_read   += (uint64_t)(bytes))
#define MEM_WRITE(bytes)  (G_STATS[TL_CUR_STAGE].bytes_written+= (uint64_t)(bytes))
#endif

struct StageTimer {
    Stage s; std::chrono::high_resolution_clock::time_point t0;
    StageTimer(Stage st) : s(st), t0(std::chrono::high_resolution_clock::now()) { TL_CUR_STAGE = s; G_STATS[s].calls++; }
    ~StageTimer() {
        auto t1 = std::chrono::high_resolution_clock::now();
        long double ms = std::chrono::duration<long double, std::milli>(t1 - t0).count();
        G_STATS[s].ms_total.store(G_STATS[s].ms_total.load() + ms);
        TL_CUR_STAGE = ST_MISC;
    }
};

static inline void print_stats_summary(const char* header) {
    printf("\r\n===== %s =====\r\n", header);
    for (int s = 0; s < ST_COUNT; ++s) {
        long double ms = G_STATS[s].ms_total.load();
        double sec = (double)ms / 1000.0;
        double rd_MB = (double)G_STATS[s].bytes_read.load() / (1024.0 * 1024.0);
        double wr_MB = (double)G_STATS[s].bytes_written.load() / (1024.0 * 1024.0);
        double rd_MBps = (sec > 0.0) ? (rd_MB / sec) : 0.0;
        double wr_MBps = (sec > 0.0) ? (wr_MB / sec) : 0.0;
        printf("Stage %-10s | calls:%4llu | time:%9.3Lf ms | R:%7.2f MB | W:%7.2f MB | R:%8.2f MB/s | W:%8.2f MB/s\r\n",
            STAGE_NAME[s],
            (unsigned long long)G_STATS[s].calls.load(),
            ms, rd_MB, wr_MB, rd_MBps, wr_MBps
        );
    }
    fflush(stdout);
}

static Image* readBMP(const char* filename) {
    StageTimer _t(ST_IO);
    FILE* fp = fopen(filename, "rb"); if (!fp) return nullptr;
    BITMAPFILEHEADER bfh{}; BITMAPINFOHEADER bih{};
    fread(&bfh, sizeof(bfh), 1, fp); MEM_READ(sizeof(bfh));
    fread(&bih, sizeof(bih), 1, fp); MEM_READ(sizeof(bih));
    Image* img = (Image*)malloc(sizeof(Image)); img->width = bih.biWidth; img->height = abs(bih.biHeight); img->channels = 3;
    int row_padded = ((img->width * 3 + 3) / 4) * 4; img->data = (unsigned char*)malloc((size_t)img->width * img->height * 3);
    fseek(fp, bfh.bfOffBits, SEEK_SET);
    for (int i = 0; i < img->height; ++i) { size_t got = fread(img->data + (size_t)i * img->width * 3, 1, (size_t)img->width * 3, fp); MEM_READ(got); fseek(fp, row_padded - img->width * 3, SEEK_CUR); } fclose(fp); return img;
}

static int writeBMP(const char* filename, const Image* img) {
    StageTimer _t(ST_IO);
    FILE* fp = fopen(filename, "wb"); if (!fp) return 0;
    int w = img->width, h = img->height; int c = img->channels; if (c != 3) c = 3;
    int row_padded = ((w * 3 + 3) / 4) * 4; int fileSize = 54 + row_padded * h;
    unsigned char bmpfileheader[14] = { 'B','M',0,0,0,0,0,0,0,0,54,0,0,0 };
    unsigned char bmpinfoheader[40] = { 0 }; unsigned char pad[3] = { 0,0,0 };
    bmpfileheader[2] = (unsigned char)(fileSize);
    bmpfileheader[3] = (unsigned char)(fileSize >> 8);
    bmpfileheader[4] = (unsigned char)(fileSize >> 16);
    bmpfileheader[5] = (unsigned char)(fileSize >> 24);
    bmpinfoheader[0] = 40;
    bmpinfoheader[4] = (unsigned char)(w);
    bmpinfoheader[5] = (unsigned char)(w >> 8);
    bmpinfoheader[6] = (unsigned char)(w >> 16);
    bmpinfoheader[7] = (unsigned char)(w >> 24);
    bmpinfoheader[8] = (unsigned char)(h);
    bmpinfoheader[9] = (unsigned char)(h >> 8);
    bmpinfoheader[10] = (unsigned char)(h >> 16);
    bmpinfoheader[11] = (unsigned char)(h >> 24);
    bmpinfoheader[12] = 1; bmpinfoheader[14] = 24;
    fwrite(bmpfileheader, 1, 14, fp); MEM_WRITE(14);
    fwrite(bmpinfoheader, 1, 40, fp); MEM_WRITE(40);

    for (int i = 0; i < h; ++i) {
        const unsigned char* s = &img->data[(size_t)i * w * img->channels];
        fwrite(s, 1, (size_t)w * 3, fp); MEM_WRITE(w * 3);
        fwrite(pad, 1, (size_t)(row_padded - w * 3), fp); MEM_WRITE(row_padded - w * 3);
    }
    fclose(fp); return 1;
}

// ===================== Tiny Thread Pool =====================
class TinyPool {
public:
    explicit TinyPool(unsigned n = std::thread::hardware_concurrency()) : stop(false) { n = n ? n : 1; for (unsigned i = 0; i < n; ++i) workers.emplace_back([this] { run(); }); }
    ~TinyPool() { { std::unique_lock<std::mutex> lk(mx); stop = true; cv.notify_all(); } for (auto& t : workers) t.join(); }
    void submit(std::function<void()> fn) { { std::unique_lock<std::mutex> lk(mx); q.emplace(std::move(fn)); } cv.notify_one(); }
    void wait_idle() { std::unique_lock<std::mutex> lk(mx); done_cv.wait(lk, [&] { return q.empty() && active == 0; }); }
private:
    void run() { for (;;) { std::function<void()> job; { std::unique_lock<std::mutex> lk(mx); cv.wait(lk, [&] { return stop || !q.empty(); }); if (stop && q.empty()) return; job = std::move(q.front()); q.pop(); ++active; } job(); { std::unique_lock<std::mutex> lk(mx); --active; if (q.empty() && active == 0) done_cv.notify_all(); } } }
    std::vector<std::thread> workers; std::mutex mx; std::condition_variable cv, done_cv; std::queue<std::function<void()>> q; std::atomic<bool> stop; int active = 0;
};

// ===================== LUT =====================
#pragma pack(push,1)
struct MapEntry { int16_t x0, y0; uint16_t wx, wy; }; // 8 bytes
#pragma pack(pop)
static inline int clampi(int v, int lo, int hi) { return v < lo ? lo : (v > hi ? hi : v); }

// ===== Faster single-thread LUT builder (drop-in) =====
static inline int clampi_fast(int v, int lo, int hi) {
    // branchless-ish clamp
    if (v < lo) v = lo;
    if (v > hi) v = hi;
    return v;
}

static void buildCompositeLUT(
    std::vector<MapEntry>& lut,
    const std::vector<int16_t>& mapx_q,
    const std::vector<int16_t>& mapy_q,
    int in_w, int in_h,
    int resized_w, int resized_h,
    int crop_w, int crop_h,
    int crop_x0, int crop_y0,
    float roll_deg, int y_shift_px,
    float resize_factor_local
) {
    StageTimer _t(ST_LUT);
    lut.resize((size_t)crop_w * crop_h);

    // Precompute constants once
    const int32_t cxr_q = (resized_w << 15);
    const int32_t cyr_q = (resized_h << 15);

    const float ang = roll_deg * (float)M_PI / 180.0f;
    const int32_t cos_q = (int32_t)llroundf(cosf(ang) * (1 << WFIX_SHIFT));
    const int32_t sin_q = (int32_t)llroundf(sinf(ang) * (1 << WFIX_SHIFT));

    const int32_t yshift_q = ((int32_t)y_shift_px) << WFIX_SHIFT;
    const int32_t inv_resize_q = (int32_t)llroundf((1.0f / resize_factor_local) * (1 << WFIX_SHIFT));

    const int32_t max_x_q = ((in_w - 2) << WFIX_SHIFT);
    const int32_t max_y_q = ((in_h - 2) << WFIX_SHIFT);

    // Walk rows with raw pointers to avoid repeated indexing
    for (int y = 0; y < crop_h; ++y) {
        const int tgt_y = crop_y0 + y;
        const size_t base = (size_t)tgt_y * resized_w + crop_x0;

        const int16_t* __restrict mx = &mapx_q[base];
        const int16_t* __restrict my = &mapy_q[base];
        MapEntry* __restrict me = &lut[(size_t)y * crop_w];

        // light unrolling by 2 gives MSVC a good hint without hurting icache
        int x = 0;
        for (; x + 1 < crop_w; x += 2) {
#pragma clang loop unroll(enable)
            for (int u = 0; u < 2; ++u) {
                // load source (Q12.4 -> Q16.16)
                const int32_t srcx_q16 = ((int32_t)mx[u]) << (WFIX_SHIFT - MAP_SHIFT);
                const int32_t srcy_q16 = ((int32_t)my[u]) << (WFIX_SHIFT - MAP_SHIFT);

                // translate to center and apply vertical shift
                const int32_t dx_q = srcx_q16 - cxr_q;
                const int32_t dy_q = (srcy_q16 - cyr_q) - yshift_q;

                // rotate (Q16.16), keep 64-bit only at the multiply
                const int32_t ax_q = (int32_t)(((int64_t)cos_q * dx_q + (int64_t)sin_q * dy_q) >> WFIX_SHIFT) + cxr_q;
                const int32_t ay_q = (int32_t)((-(int64_t)sin_q * dx_q + (int64_t)cos_q * dy_q) >> WFIX_SHIFT) + cyr_q;

                // inverse resize
                int32_t ox_q = (int32_t)(((int64_t)ax_q * inv_resize_q) >> WFIX_SHIFT);
                int32_t oy_q = (int32_t)(((int64_t)ay_q * inv_resize_q) >> WFIX_SHIFT);

                // clamp once in fixed space (branchless-ish)
                ox_q = clampi_fast(ox_q, 0, max_x_q);
                oy_q = clampi_fast(oy_q, 0, max_y_q);

                MapEntry e;
                e.x0 = (int16_t)(ox_q >> WFIX_SHIFT);
                e.y0 = (int16_t)(oy_q >> WFIX_SHIFT);
                e.wx = (uint16_t)(ox_q & 0xFFFF);
                e.wy = (uint16_t)(oy_q & 0xFFFF);
                me[u] = e; MEM_WRITE(sizeof(MapEntry));
            }
            mx += 2; my += 2; me += 2;
        }

        // tail
        for (; x < crop_w; ++x) {
            const int32_t srcx_q16 = ((int32_t)*mx++) << (WFIX_SHIFT - MAP_SHIFT);
            const int32_t srcy_q16 = ((int32_t)*my++) << (WFIX_SHIFT - MAP_SHIFT);

            const int32_t dx_q = srcx_q16 - cxr_q;
            const int32_t dy_q = (srcy_q16 - cyr_q) - yshift_q;

            const int32_t ax_q = (int32_t)(((int64_t)cos_q * dx_q + (int64_t)sin_q * dy_q) >> WFIX_SHIFT) + cxr_q;
            const int32_t ay_q = (int32_t)((-(int64_t)sin_q * dx_q + (int64_t)cos_q * dy_q) >> WFIX_SHIFT) + cyr_q;

            int32_t ox_q = (int32_t)(((int64_t)ax_q * inv_resize_q) >> WFIX_SHIFT);
            int32_t oy_q = (int32_t)(((int64_t)ay_q * inv_resize_q) >> WFIX_SHIFT);

            ox_q = clampi_fast(ox_q, 0, max_x_q);
            oy_q = clampi_fast(oy_q, 0, max_y_q);

            MapEntry e;
            e.x0 = (int16_t)(ox_q >> WFIX_SHIFT);
            e.y0 = (int16_t)(oy_q >> WFIX_SHIFT);
            e.wx = (uint16_t)(ox_q & 0xFFFF);
            e.wy = (uint16_t)(oy_q & 0xFFFF);
            *me++ = e; MEM_WRITE(sizeof(MapEntry));
        }
    }
}


// ===================== Renderer (scalar, RGB) =====================
static void renderFromLUT(const Image* src, const std::vector<MapEntry>& lut, int out_w, int out_h, Image* out) {
    StageTimer _t(ST_RENDER);
    out->width = out_w; out->height = out_h; out->channels = 3;
    out->data = (unsigned char*)malloc((size_t)out_w * out_h * 3);
    const int stride = src->width * 3; // RGB

    for (int y = 0; y < out_h; ++y) {
        const MapEntry* row = &lut[(size_t)y * out_w];
        unsigned char* dst = &out->data[(size_t)y * out_w * 3];
        for (int x = 0; x < out_w; ++x) {
            const MapEntry e = row[x];
            int x0 = e.x0, y0 = e.y0; unsigned wx = e.wx, wy = e.wy;
            unsigned W00 = (unsigned)(((uint64_t)(0x10000 - wx) * (0x10000 - wy)) >> 16);
            unsigned W10 = (unsigned)(((uint64_t)wx * (0x10000 - wy)) >> 16);
            unsigned W01 = (unsigned)(((uint64_t)(0x10000 - wx) * wy) >> 16);
            unsigned W11 = (unsigned)(((uint64_t)wx * wy) >> 16);

            const unsigned char* p0 = &src->data[(size_t)y0 * stride + x0 * 3];
            const unsigned char* p1 = p0 + 3;              // (x0+1,y0)
            const unsigned char* p2 = p0 + stride;         // (x0,y0+1)
            const unsigned char* p3 = p2 + 3;              // (x0+1,y0+1)

            dst[0] = (unsigned char)((p0[0] * W00 + p1[0] * W10 + p2[0] * W01 + p3[0] * W11) >> 16);
            dst[1] = (unsigned char)((p0[1] * W00 + p1[1] * W10 + p2[1] * W01 + p3[1] * W11) >> 16);
            dst[2] = (unsigned char)((p0[2] * W00 + p1[2] * W10 + p2[2] * W01 + p3[2] * W11) >> 16);
            dst += 3;
        }
    }
}

// ===================== Renderer (tiled, MT, RGB) =====================
// Tile renderer improves cache locality and approximates a line-buffer:
// each tile touches a compact band of source rows.
// You can tune TILE_W / TILE_H to your CPU cache & memory.
#ifndef TILE_W
#define TILE_W  128
#endif
#ifndef TILE_H
#define TILE_H  64
#endif

struct TileJob { int x0, y0, w, h; };

static void renderFromLUT_Tiled(const Image* src,
    const std::vector<MapEntry>& lut,
    int out_w, int out_h,
    Image* out)
{
    StageTimer _t(ST_RENDER);

    out->width = out_w; out->height = out_h; out->channels = 3;
    out->data = (unsigned char*)malloc((size_t)out_w * out_h * 3);

    const int stride = src->width * 3; // RGB
    std::vector<TileJob> jobs;
    jobs.reserve(((out_w + TILE_W - 1) / TILE_W) * ((out_h + TILE_H - 1) / TILE_H));

    // Build tile grid
    for (int ty = 0; ty < out_h; ty += TILE_H) {
        int th = std::min(TILE_H, out_h - ty);
        for (int tx = 0; tx < out_w; tx += TILE_W) {
            int tw = std::min(TILE_W, out_w - tx);
            jobs.push_back({ tx, ty, tw, th });
        }
    }

    // Threaded tile processing
    TinyPool pool(std::max(1u, std::thread::hardware_concurrency()));
    for (const auto& jb : jobs) {
        pool.submit([&, jb] {
            const int x0 = jb.x0, y0 = jb.y0;
            const int tw = jb.w, th = jb.h;

            for (int y = 0; y < th; ++y) {
                const int oy = y0 + y;
                const MapEntry* lut_row = &lut[(size_t)oy * out_w + x0];
                unsigned char* dst = &out->data[(size_t)oy * out_w * 3 + x0 * 3];

                int x = 0;

                // light unroll by 2
                for (; x + 1 < tw; x += 2) {
                    // pixel #0
                    {
                        const MapEntry e = lut_row[x + 0];
                        const int x0i = e.x0, y0i = e.y0;
                        const unsigned wx = e.wx, wy = e.wy;

                        const unsigned W00 = (unsigned)(((uint64_t)(0x10000 - wx) * (0x10000 - wy)) >> 16);
                        const unsigned W10 = (unsigned)(((uint64_t)wx * (0x10000 - wy)) >> 16);
                        const unsigned W01 = (unsigned)(((uint64_t)(0x10000 - wx) * wy) >> 16);
                        const unsigned W11 = (unsigned)(((uint64_t)wx * wy) >> 16);

                        const unsigned char* p0 = &src->data[(size_t)y0i * stride + x0i * 3];
                        const unsigned char* p1 = p0 + 3;
                        const unsigned char* p2 = p0 + stride;
                        const unsigned char* p3 = p2 + 3;

                        dst[0] = (unsigned char)((p0[0] * W00 + p1[0] * W10 + p2[0] * W01 + p3[0] * W11) >> 16);
                        dst[1] = (unsigned char)((p0[1] * W00 + p1[1] * W10 + p2[1] * W01 + p3[1] * W11) >> 16);
                        dst[2] = (unsigned char)((p0[2] * W00 + p1[2] * W10 + p2[2] * W01 + p3[2] * W11) >> 16);
                        dst += 3;
                    }
                    // pixel #1
                    {
                        const MapEntry e = lut_row[x + 1];
                        const int x0i = e.x0, y0i = e.y0;
                        const unsigned wx = e.wx, wy = e.wy;

                        const unsigned W00 = (unsigned)(((uint64_t)(0x10000 - wx) * (0x10000 - wy)) >> 16);
                        const unsigned W10 = (unsigned)(((uint64_t)wx * (0x10000 - wy)) >> 16);
                        const unsigned W01 = (unsigned)(((uint64_t)(0x10000 - wx) * wy) >> 16);
                        const unsigned W11 = (unsigned)(((uint64_t)wx * wy) >> 16);

                        const unsigned char* p0 = &src->data[(size_t)y0i * stride + x0i * 3];
                        const unsigned char* p1 = p0 + 3;
                        const unsigned char* p2 = p0 + stride;
                        const unsigned char* p3 = p2 + 3;

                        dst[0] = (unsigned char)((p0[0] * W00 + p1[0] * W10 + p2[0] * W01 + p3[0] * W11) >> 16);
                        dst[1] = (unsigned char)((p0[1] * W00 + p1[1] * W10 + p2[1] * W01 + p3[1] * W11) >> 16);
                        dst[2] = (unsigned char)((p0[2] * W00 + p1[2] * W10 + p2[2] * W01 + p3[2] * W11) >> 16);
                        dst += 3;
                    }
                }

                // tail
                for (; x < tw; ++x) {
                    const MapEntry e = lut_row[x];
                    const int x0i = e.x0, y0i = e.y0;
                    const unsigned wx = e.wx, wy = e.wy;

                    const unsigned W00 = (unsigned)(((uint64_t)(0x10000 - wx) * (0x10000 - wy)) >> 16);
                    const unsigned W10 = (unsigned)(((uint64_t)wx * (0x10000 - wy)) >> 16);
                    const unsigned W01 = (unsigned)(((uint64_t)(0x10000 - wx) * wy) >> 16);
                    const unsigned W11 = (unsigned)(((uint64_t)wx * wy) >> 16);

                    const unsigned char* p0 = &src->data[(size_t)y0i * stride + x0i * 3];
                    const unsigned char* p1 = p0 + 3;
                    const unsigned char* p2 = p0 + stride;
                    const unsigned char* p3 = p2 + 3;

                    dst[0] = (unsigned char)((p0[0] * W00 + p1[0] * W10 + p2[0] * W01 + p3[0] * W11) >> 16);
                    dst[1] = (unsigned char)((p0[1] * W00 + p1[1] * W10 + p2[1] * W01 + p3[1] * W11) >> 16);
                    dst[2] = (unsigned char)((p0[2] * W00 + p1[2] * W10 + p2[2] * W01 + p3[2] * W11) >> 16);
                    dst += 3;
                }
            }
            });
    }
    pool.wait_idle();
}



// ===================== Parallel Feather Blend + Resize =====================
static Image* featherBlendFitMT(const std::vector<Image*>& frames, int overlap, int target_w, int target_h) {
    StageTimer _t(ST_BLEND);
    if (frames.empty()) return nullptr;
    const int h = frames[0]->height, c = 3; // RGB
    int total_w = frames[0]->width;
    for (size_t i = 1; i < frames.size(); ++i) total_w += (frames[i]->width - overlap);

    Image* blended = (Image*)malloc(sizeof(Image));
    blended->width = total_w; blended->height = h; blended->channels = c;
    blended->data = (unsigned char*)malloc((size_t)total_w * h * c);

    // Precompute alpha ramp
    std::vector<uint16_t> alpha(overlap);
    for (int x = 0; x < overlap; ++x) alpha[x] = (uint16_t)(((uint64_t)x << 16) / (overlap ? overlap : 1));

    // Row-parallel blend
    TinyPool pool(std::max(1u, std::thread::hardware_concurrency()));
    auto worker = [&](int y0, int y1) {
        for (int y = y0; y < y1; ++y) {
            unsigned char* dstRow = &blended->data[(size_t)y * total_w * c];
            // seed with first frame row
            const Image* img0 = frames[0];
            memcpy(dstRow, &img0->data[(size_t)y * img0->width * c], (size_t)img0->width * c);
            int curx = img0->width;

            for (size_t i = 1; i < frames.size(); ++i) {
                const Image* img = frames[i];
                const unsigned char* srcRow = &img->data[(size_t)y * img->width * c];

                int ov = overlap; if (ov > img->width) ov = img->width; if (ov > curx) ov = curx;
                // blend overlap
                for (int x = 0; x < ov; ++x) { unsigned a = alpha[x]; size_t dp = (size_t)(curx - ov + x) * c, sp = (size_t)x * c; for (int ch = 0; ch < c; ++ch) { unsigned prev = dstRow[dp + ch], curr = srcRow[sp + ch]; dstRow[dp + ch] = (unsigned char)((prev * (0x10000 - a) + curr * a) >> 16); } }
                // copy tail
                int tail = img->width - ov; if (tail > 0) memcpy(&dstRow[(size_t)curx * c], &srcRow[(size_t)ov * c], (size_t)tail * c);
                curx += tail;
            }
        }
        };

    const int chunk = 128; for (int y0 = 0; y0 < h; y0 += chunk) pool.submit([&, y0] { worker(y0, std::min(h, y0 + chunk)); }); pool.wait_idle();

    // === Resize to target_w using row-parallel bilinear (scalar fast path) ===
    StageTimer _r(ST_RESIZE);
    float scale = (float)target_w / (float)blended->width; int new_w = target_w; int new_h = (int)lroundf(blended->height * scale);

    Image* resized = (Image*)malloc(sizeof(Image)); resized->width = new_w; resized->height = new_h; resized->channels = c; resized->data = (unsigned char*)malloc((size_t)new_w * new_h * c);

    const int32_t inv_scale_q = (int32_t)llroundf((1.0f / scale) * (1 << 16));

    // Precompute sx/wx for all x
    std::vector<int> sx_arr(new_w);
    std::vector<uint16_t> wx_arr(new_w);
    for (int x = 0; x < new_w; ++x) { int32_t sx_q = ((int32_t)x << 16); sx_q = (int32_t)(((int64_t)sx_q * inv_scale_q) >> 16); int sx = clampi(sx_q >> 16, 0, blended->width - 2); sx_arr[x] = sx; wx_arr[x] = (uint16_t)(sx_q & 0xFFFF); }

    TinyPool pool2(std::max(1u, std::thread::hardware_concurrency()));
    auto resize_worker = [&](int y0, int y1) {
        for (int y = y0; y < y1; ++y) {
            int32_t sy_q = ((int32_t)y << 16); sy_q = (int32_t)(((int64_t)sy_q * inv_scale_q) >> 16);
            int sy = clampi(sy_q >> 16, 0, blended->height - 2); uint16_t wy = (uint16_t)(sy_q & 0xFFFF);
            unsigned W_y0 = 0x10000 - wy; unsigned W_y1 = wy;
            const unsigned char* row0 = &blended->data[(size_t)sy * blended->width * c];
            const unsigned char* row1 = row0 + blended->width * c;
            unsigned char* drow = &resized->data[(size_t)y * new_w * c];

            for (int x = 0; x < new_w; ++x) {
                int sx = sx_arr[x]; uint16_t wx = wx_arr[x];
                unsigned W_x0 = 0x10000 - wx; unsigned W_x1 = wx;
                unsigned W00 = (unsigned)(((uint64_t)W_x0 * W_y0) >> 16);
                unsigned W10 = (unsigned)(((uint64_t)W_x1 * W_y0) >> 16);
                unsigned W01 = (unsigned)(((uint64_t)W_x0 * W_y1) >> 16);
                unsigned W11 = (unsigned)(((uint64_t)W_x1 * W_y1) >> 16);

                const unsigned char* p0 = &row0[sx * c];
                const unsigned char* p1 = p0 + c;
                const unsigned char* p2 = &row1[sx * c];
                const unsigned char* p3 = p2 + c;

                drow[x * c + 0] = (unsigned char)((p0[0] * W00 + p1[0] * W10 + p2[0] * W01 + p3[0] * W11) >> 16);
                drow[x * c + 1] = (unsigned char)((p0[1] * W00 + p1[1] * W10 + p2[1] * W01 + p3[1] * W11) >> 16);
                drow[x * c + 2] = (unsigned char)((p0[2] * W00 + p1[2] * W10 + p2[2] * W01 + p3[2] * W11) >> 16);
            }
        }
        };

    const int chunk2 = 128; for (int y0 = 0; y0 < new_h; y0 += chunk2) pool2.submit([&, y0] { resize_worker(y0, std::min(new_h, y0 + chunk2)); }); pool2.wait_idle();

    // Fit to final target (center crop/pad vertically)
    Image* finalOut = (Image*)malloc(sizeof(Image)); finalOut->width = target_w; finalOut->height = target_h; finalOut->channels = c; finalOut->data = (unsigned char*)calloc((size_t)target_w * target_h * c, 1);
    int yoff = (target_h - new_h) / 2; if (yoff < 0) yoff = 0; int copy_h = (new_h > target_h) ? target_h : new_h;
    for (int y = 0; y < copy_h; ++y) { if (y + yoff < 0 || y + yoff >= target_h) continue; memcpy(&finalOut->data[(size_t)(y + yoff) * target_w * c], &resized->data[(size_t)y * new_w * c], (size_t)new_w * c); }

    free(resized->data); free(resized);
    free(blended->data); free(blended);
    return finalOut;
}

// ===================== Sensor to camera =====================
static void convert_rpy_sensor_to_cam(double roll_sys1, double pitch_sys1, double yaw_offset, double pitch_offset, double roll_offset, float* roll_deg_out, float* pitch_deg_out, float* yaw_deg_out) { StageTimer _t(ST_MISC); double rr = roll_sys1 * M_PI / 180.0, pp = pitch_sys1 * M_PI / 180.0; double ror = roll_offset * M_PI / 180.0, por = pitch_offset * M_PI / 180.0, yor = yaw_offset * M_PI / 180.0; double Rx[3][3] = { {1,0,0},{0,cos(rr),-sin(rr)},{0,sin(rr),cos(rr)} }; double Ry[3][3] = { {cos(pp),0,sin(pp)},{0,1,0},{-sin(pp),0,cos(pp)} }; double R_sys1[3][3]; for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j) R_sys1[i][j] = Rx[i][0] * Ry[0][j] + Rx[i][1] * Ry[1][j] + Rx[i][2] * Ry[2][j]; double Rr[3][3] = { {1,0,0},{0,cos(ror),-sin(ror)},{0,sin(ror),cos(ror)} }; double Rp[3][3] = { {cos(por),0,sin(por)},{0,1,0},{-sin(por),0,cos(por)} }; double Rz[3][3] = { {cos(yor),-sin(yor),0},{sin(yor),cos(yor),0},{0,0,1} }; double Rt[3][3]; for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j) Rt[i][j] = Rz[i][0] * Rp[0][j] + Rz[i][1] * Rp[1][j] + Rz[i][2] * Rp[2][j]; double Roff[3][3]; for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j) Roff[i][j] = Rt[i][0] * Rr[0][j] + Rt[i][1] * Rr[1][j] + Rt[i][2] * Rr[2][j]; double R[3][3]; for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j) R[i][j] = R_sys1[i][0] * Roff[0][j] + R_sys1[i][1] * Roff[1][j] + R_sys1[i][2] * Roff[2][j]; double pitchX = asin(-R[2][0]); double cy = cos(pitchX); double rollX, yawX; if (fabs(cy) > 1e-6) { rollX = atan2(R[2][1], R[2][2]); yawX = atan2(R[1][0], R[0][0]); } else { rollX = 0; yawX = atan2(-R[0][1], R[1][1]); } *roll_deg_out = (float)(rollX * 180.0 / M_PI); *pitch_deg_out = (float)(pitchX * 180.0 / M_PI); *yaw_deg_out = (float)(yawX * 180.0 / M_PI); }
